% Airfoil shape command used in the 3D view
\newcommand{\drawAirfoil}[1]{%
  plot [smooth cycle, tension=0.6] coordinates {%
    (#1, 0, 0)%
    (#1, 0.2*\chord, 0.1*\chord)%
    (#1, 0.6*\chord, 0.08*\chord)%
    (#1, \chord, 0)%
    (#1, 0.6*\chord, -0.05*\chord)%
    (#1, 0.2*\chord, -0.08*\chord)%
  }%
}

% ---- CROSS-SECTION (TYPICAL SECTION) AS A MACRO ----
\newcommand{\WingCrossSection}{%
\begin{tikzpicture}[scale=1.5, >=Latex]
  % Airfoil shape
  \draw[thick, fill=UofSCLightGray] plot [smooth cycle, tension=0.6] coordinates {
    (0,0)           % LE
    (1.2, 0.6)      % Upper 1
    (3.6, 0.48)     % Upper 2
    (6,0)           % TE
    (3.6, -0.3)     % Lower 2
    (1.2, -0.48)    % Lower 1
  };

  % Reference chord line
  \draw[dashed, UofSCBlack] (-0.2,0) -- (6.5,0) node[right] {Reference Chord Line};

  % Points
  \coordinate (Q) at (1.2,0);
  \coordinate (C) at (2.4,0);
  \coordinate (P) at (3.6,0);

  % Distances
  \draw[<->, thick] ($(Q)+(0,1)$) -- ($(P)+(0,1)$) node[midway, fill=white] {$x_{QP}$};
  \draw[dashed] (P) -- ($(P)+(0,1.2)$);
  \draw[<->, thick] ($(P)+(0,0.75)$) -- ($(C)+(0,0.75)$) node[midway, fill=white] {$x_{CP}$};
  \draw[dashed] (C) -- ($(C)+(0,0.8)$);

  % Points markers
  \fill[UofSCGarnet] (P) circle (2pt) node[below=7mm] {P};
  \fill[UofSCBlack] (C) circle (2pt) node[below=7mm] {C};
  \fill[UofSCAtlantic] (Q) circle (2pt) node[below=7mm] {Q};

  % Freestream velocity
  \draw[->, ultra thick, UofSCAtlantic] (-2, 0) -- (-0.5, 0) node[above] {$U$};

  % Lift at Q
  \draw[->, ultra thick, UofSCAtlantic] (Q) -- ($(Q)+(0,1.5)$) node[right] {$L(U)$};

  % Pitch DOF
  \draw[->, ultra thick, UofSCGarnet] ($(Q)+(-2,-0.4)$) arc (270:90:0.4)
    node[midway, above left] {$\theta(z,t)$};
\end{tikzpicture}%
}

% ---- 3D ISOMETRIC WING VIEW AS A MACRO ----
\newcommand{\WingThreeD}{%
\begin{tikzpicture}[x={(0.866cm,-0.4cm)}, y={(0.6cm,0.5cm)}, z={(0cm,1cm)}, scale=1.3]

  % Constants
  \def\span{6}   % Length along z axis (visual span)
  \def\chord{3}  % Chord length

  % Wall (Root)
  \fill[UofSCWarmGrey!60] (0, -1, -2) -- (0, \chord+1, -2) --
                           (0, \chord+1, 2) -- (0, -1, 2) -- cycle;
  \draw[thick] (0, -1, -2) -- (0, \chord+1, -2) --
               (0, \chord+1, 2) -- (0, -1, 2) -- cycle;

  % Bottom surface lines (root to tip)
  \draw[thick, UofSCBlack] (0, \chord, 0) -- (\span, \chord, 0); % Trailing edge
  \draw[thick, UofSCBlack] (0, 0, 0)     -- (\span, 0, 0);       % Leading edge

  % Tip profile (visible cross section)
  \draw[thick, fill=UofSC10Black] \drawAirfoil{\span};

  % Root profile (at wall) - dashed
  \draw[dashed, thick] \drawAirfoil{0};

  % Internal axes along the span
  % Q (Aero axis) at 20% chord
  \coordinate (RootQ) at (0, 0.2*\chord, 0);
  \coordinate (TipQ)  at (\span, 0.2*\chord, 0);
  \draw[densely dotted, UofSCAtlantic, thick] (RootQ) -- (TipQ);
  \node[right, font=\footnotesize, UofSCAtlantic]
    at (\span+0.3, 0.2*\chord, 0) {$Q_{\text{axis}}$};
  \fill[UofSCAtlantic] (TipQ) circle (1pt);

  % C (Mass axis) at 40% chord
  \coordinate (RootC) at (0, 0.4*\chord, 0);
  \coordinate (TipC)  at (\span, 0.4*\chord, 0);
  \draw[densely dotted, UofSCBlack, thick] (RootC) -- (TipC);
  \node[right, font=\footnotesize, UofSCBlack]
    at (\span+0.3, 0.4*\chord, 0) {$C_{\text{axis}}$};
  \fill[UofSCBlack] (TipC) circle (1pt);

  % P (Elastic axis) at 60% chord
  \coordinate (RootP) at (0, 0.6*\chord, 0);
  \coordinate (TipP)  at (\span, 0.6*\chord, 0);
  \draw[dashed, UofSCGarnet, thick] (RootP) -- (TipP);
  \node[right, font=\footnotesize, UofSCGarnet]
    at (\span+0.3, 0.6*\chord, 0) {$P_{\text{axis}}$};
  \fill[UofSCGarnet] (TipP) circle (1pt);

  % Wing side surface shading
  \fill[UofSCAtlantic!40, opacity=0.9]
    plot [smooth, tension=0.6] coordinates {
      (0,0,0) (0, 0.2*\chord, 0.1*\chord)
      (0, 0.6*\chord, 0.08*\chord) (0,\chord,0)
    }
    --
    plot [smooth, tension=0.6] coordinates {
      (\span,\chord,0) (\span, 0.6*\chord, 0.08*\chord)
      (\span, 0.2*\chord, 0.1*\chord) (\span,0,0)
    }
    -- cycle;

  % Leading and trailing edges
  \draw[thick] (0,0,0)     -- (\span,0,0);
  \draw[thick] (0,\chord,0) -- (\span,\chord,0);

  % Tip profile (outline)
  \draw[thick] \drawAirfoil{\span};

  % Spanwise axis
  \draw[->, thick, UofSCGarnet] (0, \chord, 0) --
    (\span + 0.5, \chord, 0) node[right] {$z$ (Span)};

  % Freestream velocity U (multiple arrows)
  \foreach \x in {1, 2, 3, 4, 5} {
    \draw[->, ultra thick, UofSCGarnet] (\x, -1.2, 0) -- (\x, -0.2, 0);
  }
  \node[right, UofSCGarnet] at (3, -2.0, 0) {$U$};

  % Labels
  \node[align=center] at (\span+0.5, \chord+0.5, 0.5) {Tip (Free)};
  \node[align=center] at (-0.5, \chord/2, 2.5) {Root (Fixed)};

\end{tikzpicture}%
}
