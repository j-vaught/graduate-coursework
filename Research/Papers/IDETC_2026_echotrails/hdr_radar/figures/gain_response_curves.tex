\documentclass[border=2pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage{xcolor}

% Brand colors
\definecolor{garnet}{HTML}{73000A}
\definecolor{atlantic}{HTML}{466A9F}
\definecolor{congaree}{HTML}{1F414D}
\definecolor{horseshoe}{HTML}{65780B}
\definecolor{black90}{HTML}{363636}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=\columnwidth,
    height=0.65\columnwidth,
    xlabel={Receiver Gain Setting},
    ylabel={Digitized Intensity (4-bit)},
    xmin=2, xmax=99,
    ymin=-0.5, ymax=17.5,
    ytick={0,2,4,6,8,10,12,14,16},
    grid=major,
    grid style={black30, opacity=0.3},
    axis line style={black90},
    tick label style={color=black90},
    label style={color=black90},
    legend pos=south east,
    legend style={
        font=\footnotesize,
        draw=black30,
        fill opacity=0.9,
        text opacity=1,
    },
    every axis plot/.append style={thick, no marks},
    clip=true,
]

\addplot[black30, densely dotted, thin]
    coordinates {(2,16) (99,16)};
\addlegendentry{4-bit ceiling}

% Strong buoy
\addplot[garnet, line width=1.6pt]

    {gain_response_data/strong_buoy.csv};
\addlegendentry{Strong buoy}

% Weak buoy
\addplot[atlantic, line width=1.6pt]
    table[col sep=comma, x=gain, y=intensity]
    {gain_response_data/moderate_buoy.csv};

% Moderate buoy
\addplot[horseshoe, line width=1.6pt]
    table[col sep=comma, x=gain, y=intensity]
    {gain_response_data/weak_buoy.csv};

% Land (diffuse)
\addplot[congaree, line width=1.6pt, densely dashed]

\end{axis}
