\documentclass[border=4pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, fit, backgrounds}
\usepackage{xcolor}

% Brand colors
\definecolor{garnet}{HTML}{73000A}
\definecolor{atlantic}{HTML}{466A9F}
\definecolor{congaree}{HTML}{1F414D}
\definecolor{horseshoe}{HTML}{65780B}
\definecolor{black90}{HTML}{363636}
\definecolor{black70}{HTML}{5C5C5C}
\definecolor{black30}{HTML}{C7C7C7}
\definecolor{black10}{HTML}{ECECEC}

\begin{document}
\begin{tikzpicture}[
    >=Stealth,
    box/.style={
        draw=black90,
        thick,
        minimum width=3.4cm,
        minimum height=1.5cm,
        align=center,
        font=\sffamily\small,
        text=black90,
    },
    smallbox/.style={
        draw=black70,
        thick,
        minimum width=2.2cm,
        minimum height=0.7cm,
        align=center,
        font=\sffamily\footnotesize,
        text=black90,
    },
    lbl/.style={
        font=\sffamily\scriptsize,
        text=black70,
    },
]

% --- Top row: Radar and Power side by side ---
\node[box, fill=garnet!8] (radar) {
    \textbf{Furuno DRS4D-NXT}\\[2pt]
    X-Band Radar\\
    {\footnotesize(Radome Unit)}
};

\node[box, fill=black10, right=1.8cm of radar] (power) {
    \textbf{Power Supply}\\[2pt]
    {\footnotesize 12V DC}
};

% --- Middle: Raspberry Pi spanning below both ---
\coordinate (midpoint) at ($(radar.south)!0.5!(power.south)$);
\node[box, fill=atlantic!8, minimum width=5cm, below=1.4cm of midpoint] (rpi) {
    \textbf{Raspberry Pi}\\[2pt]
    Controller
};

% --- Hardware arrows ---
% Radar <-> Pi: vertical arrows offset left/right
\draw[->, thick, garnet]
    ([xshift=6pt]radar.south) -- node[lbl, right, align=left] {4-bit\\echo data} ([xshift=-12pt]rpi.north);
\draw[->, thick, atlantic]
    ([xshift=-18pt]rpi.north) -- node[lbl, left, align=right] {gain/PW\\commands} ([xshift=-6pt]radar.south);

% Power -> Pi
\draw[->, thick, black70]
    (power.south) -- node[lbl, right] {12V} ([xshift=12pt]rpi.north);

% --- Software loop below Pi ---
\node[smallbox, fill=congaree!8, below=1.8cm of rpi, xshift=-2.6cm] (read) {
    1. Read scan\\[-1pt]
    {\scriptsize(4-bit data)}
};
\node[smallbox, fill=congaree!8, right=0.4cm of read] (decide) {
    2. Decide next\\[-1pt]
    {\scriptsize gain setting}
};
\node[smallbox, fill=congaree!8, right=0.4cm of decide] (send) {
    3. Send\\[-1pt]
    {\scriptsize command}
};
\node[smallbox, fill=congaree!8, below=0.6cm of read, xshift=0cm] (log) {
    4. Log \&\\[-1pt]
    {\scriptsize process data}
};

% Software arrows: Read -> Decide -> Send, then Send curves down to Log, Log curves back to Read
\draw[->, thick, congaree] (read.east) -- (decide.west);
\draw[->, thick, congaree] (decide.east) -- (send.west);
\draw[->, thick, congaree] (send.south) |- (log.east);
\draw[->, thick, congaree] (log.north) -- (read.south);

% Software loop background
\begin{scope}[on background layer]
\node[
    draw=congaree,
    dashed,
    thick,
    fit=(read)(decide)(send)(log),
    inner sep=10pt,
    fill=congaree!3,
    label={[lbl, text=congaree, font=\sffamily\footnotesize\bfseries]above:Software Loop},
] (swloop) {};
\end{scope}

% Connection from Pi to software loop
\draw[->, thick, dashed, black70]
    (rpi.south) -- (swloop.north);

\end{tikzpicture}
\end{document}
