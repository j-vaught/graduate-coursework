% T1: moves from upper-left toward lower-right
%   Target at (-1.6,  1.4), direction unit vector ~ (0.94, -0.34)  (angle -20 deg from horizontal)
%   Trail dots step backward: (-0.94, +0.34) per step, spacing 0.12cm
%
% T2: moves from lower-left toward upper-right
%   Target at (-1.4, -1.5), direction unit vector ~ (0.87,  0.50)  (angle +30 deg from horizontal)
%   Trail dots step backward: (-0.87, -0.50) per step, spacing 0.12cm
%
% Crossing angle between directions: ~50 deg -- shallow enough for extended overlap
% Geometric intersection of the two trajectory lines (unchanged by spacing):
%   Line T1: (-1.6 + t*0.9397, 1.4 + t*(-0.3420))
%   Line T2: (-1.4 + s*0.8660, -1.5 + s*0.5000)
%   Intersection at approximately (1.60, 0.23)

\begin{tikzpicture}

% Black background square
\fill[black] (-3.2,-3.2) rectangle (3.2,3.2);
\draw[Black70, line width=0.4pt] (-3.2,-3.2) rectangle (3.2,3.2);

% Range rings: 3 concentric dashed Black50 circles
\draw[Black50, dashed, thin] (0,0) circle (1.0cm);
\draw[Black50, dashed, thin] (0,0) circle (2.0cm);
\draw[Black50, dashed, thin] (0,0) circle (3.0cm);

% White crosshair at center
\draw[white, line width=0.8pt] (-0.18,0) -- (0.18,0);
\draw[white, line width=0.8pt] (0,-0.18) -- (0,0.18);

% ============================================================
% T1: Target upper-left, moving lower-right
% Position: (-1.6, 1.4)
% Direction: (cos(-20deg), sin(-20deg)) = (0.9397, -0.3420)
% Trail goes backward from target: add (+0.9397, -0.3420)*spacing each step going toward target
% spacing = 0.12cm, 12 trail dots
% ============================================================
\def\TaxX{-1.6}
\def\TaxY{1.4}
\def\TadX{0.9397}    % direction x (forward)
\def\TadY{-0.3420}   % direction y (forward)
\def\Tspacing{0.12}

% Trail for T1 (dots behind target, i=1 is closest to target)
\foreach \i/\op in {1/0.80,2/0.68,3/0.57,4/0.47,5/0.39,6/0.32,7/0.26,8/0.21,9/0.16,10/0.12,11/0.09,12/0.06}{%
  \fill[Grass, opacity=\op]
    ({\TaxX - \i*\TadX*\Tspacing}, {\TaxY - \i*\TadY*\Tspacing})
    circle (3pt);
}
% T1 dot
\fill[Grass] (\TaxX, \TaxY) circle (4.5pt);
% T1 label
\node[Grass, font=\small\bfseries, anchor=south east] at (-1.75, 1.55) {T1};

% ============================================================
% T2: Target lower-left, moving upper-right
% Position: (-1.4, -1.5)
% Direction: (cos(30deg), sin(30deg)) = (0.8660, 0.5000)
% spacing = 0.12cm, 12 trail dots
% ============================================================
\def\TbxX{-1.4}
\def\TbxY{-1.5}
\def\TbdX{0.8660}
\def\TbdY{0.5000}

% Trail for T2
\foreach \i/\op in {1/0.80,2/0.68,3/0.57,4/0.47,5/0.39,6/0.32,7/0.26,8/0.21,9/0.16,10/0.12,11/0.09,12/0.06}{%
  \fill[Grass, opacity=\op]
    ({\TbxX - \i*\TbdX*\Tspacing}, {\TbxY - \i*\TbdY*\Tspacing})
    circle (3pt);
}
% T2 dot
\fill[Grass] (\TbxX, \TbxY) circle (4.5pt);
% T2 label
\node[Grass, font=\small\bfseries, anchor=north east] at (-1.55, -1.62) {T2};

% ============================================================
% Overlap zone annotation
% Geometric intersection of the two trajectory lines at approximately (1.60, 0.23)
% Arrow from label to overlap zone
% ============================================================

% Arrow from label to overlap zone
\draw[white, -{Stealth[length=4pt, width=3pt]}, line width=0.6pt]
  (2.1, -0.8) -- (1.65, 0.20);

% Label "Overlap zone"
\node[white, font=\tiny, anchor=north west] at (1.55, -0.72) {Overlap};
\node[white, font=\tiny, anchor=north west] at (1.55, -0.93) {zone};

\end{tikzpicture}
