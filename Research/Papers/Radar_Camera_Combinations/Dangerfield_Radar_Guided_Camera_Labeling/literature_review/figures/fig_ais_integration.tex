\begin{figure}[t]
\centering
\colorlet{garnet}{Garnet}
\colorlet{atlantic}{Atlantic}
\colorlet{congaree}{Congaree}
\colorlet{darkgray}{black!90}
\definecolor{mediumgray}{RGB}{92, 92, 92}
\definecolor{lightgray}{RGB}{235, 235, 235}
\begin{tikzpicture}[
    node distance=2cm,
    every node/.style={font=\sffamily},
    sensor_box/.style={draw, rectangle, minimum width=3cm, minimum height=1.2cm, line width=1.5pt},
    garnet_box/.style={sensor_box, draw=garnet, fill=garnet!10},
    atlantic_box/.style={sensor_box, draw=atlantic, fill=atlantic!10},
    congaree_box/.style={sensor_box, draw=congaree, fill=congaree!10},
    fusion_box/.style={draw, rectangle, minimum width=4cm, minimum height=3.5cm, line width=2pt, draw=darkgray},
    output_box/.style={draw, rectangle, minimum width=3.5cm, minimum height=2.2cm, line width=1.5pt, draw=garnet, fill=garnet!10},
    arrow/.style={->, line width=1.5pt, draw=darkgray},
    advantage_box/.style={draw=none, rectangle, fill=lightgray, minimum width=4cm, align=center, font=\small},
    limitation_box/.style={draw=none, rectangle, fill=mediumgray!20, minimum width=4cm, align=center, font=\small},
]

    % LEFT SIDE - SENSOR INPUTS
    % X-Band Radar
    \node[garnet_box] (radar) at (0, 5) {
        \begin{tabular}{c}
            \textbf{\textcolor{garnet}{X-Band Radar}} \\
            \textcolor{garnet}{Sensor 1}
        \end{tabular}
    };

    \node[anchor=west, align=left, font=\small, text width=3.5cm] (radar_specs) at (3.5, 5) {
        \textbf{\textcolor{garnet}{Provides:}}
        \begin{itemize}\setlength\itemsep{-0.3em}
            \item Range
            \item Bearing
        \end{itemize}
        \textbf{\textcolor{garnet}{Detects:}}
        \begin{itemize}\setlength\itemsep{-0.3em}
            \item All radar-reflective targets
            \item Works day/night, all-weather
        \end{itemize}
    };

    % Camera
    \node[atlantic_box] (camera) at (0, 2.5) {
        \begin{tabular}{c}
            \textbf{\textcolor{atlantic}{Camera}} \\
            \textcolor{atlantic}{Sensor 2}
        \end{tabular}
    };

    \node[anchor=west, align=left, font=\small, text width=3.5cm] (camera_specs) at (3.5, 2.5) {
        \textbf{\textcolor{atlantic}{Provides:}}
        \begin{itemize}\setlength\itemsep{-0.3em}
            \item High-resolution imagery
            \item Color, texture information
        \end{itemize}
        \textbf{\textcolor{atlantic}{Function:}}
        \begin{itemize}\setlength\itemsep{-0.3em}
            \item Substrate for annotations
        \end{itemize}
    };

    % AIS Transponder
    \node[congaree_box] (ais) at (0, 0) {
        \begin{tabular}{c}
            \textbf{\textcolor{congaree}{AIS Transponder}} \\
            \textcolor{congaree}{Sensor 3}
        \end{tabular}
    };

    \node[anchor=west, align=left, font=\small, text width=3.5cm] (ais_specs) at (3.5, 0) {
        \textbf{\textcolor{congaree}{Provides:}}
        \begin{itemize}\setlength\itemsep{-0.3em}
            \item Vessel ID, MMSI
            \item Type, Dimensions
            \item Heading, Speed
        \end{itemize}
        \textbf{\textcolor{congaree}{Limitation:}}
        \begin{itemize}\setlength\itemsep{-0.3em}
            \item Cooperative only (>300 GT)
        \end{itemize}
    };

    % CENTER - FUSION/ASSOCIATION
    \node[fusion_box] (fusion) at (9, 2.5) {
        \begin{tabular}{c}
            \Large\textbf{Sensor Fusion \& Association} \\[0.3cm]
            \hline
            \\[0.2cm]
            \textbf{WHERE}\\
            Radar: centroid projection \\
            in image coordinates \\[0.3cm]
            \textbf{WHAT SIZE}\\
            AIS: vessel length $\rightarrow$ \\
            bounding box width \\[0.3cm]
            \textbf{WHAT CLASS}\\
            AIS: vessel type code \\
            $\rightarrow$ semantic label \\[0.3cm]
            \textbf{WHERE EXACTLY}\\
            Camera: image substrate
        \end{tabular}
    };

    % Arrows from sensors to fusion
    \path[arrow] (radar.east) -- (fusion.west |- radar.east);
    \path[arrow] (camera.east) -- (fusion.west |- camera.east);
    \path[arrow] (ais.east) -- (fusion.west |- ais.east);

    % RIGHT SIDE - OUTPUT
    \node[output_box] (output) at (16.5, 2.5) {
        \begin{tabular}{c}
            \textbf{\textcolor{garnet}{Generated}} \\
            \textbf{\textcolor{garnet}{Bounding Box}} \\
            \textbf{\textcolor{garnet}{Annotation}} \\[0.3cm]
            \hline
            \\[0.2cm]
            Position: Radar proj. \\
            Width: AIS length $\times$ \\
            \hspace{0.5cm}range scaling \\
            Height: AIS type $\times$ \\
            \hspace{0.5cm}range scaling \\
            Class: AIS vessel type \\[0.2cm]
            Placed on camera frame
        \end{tabular}
    };

    % Arrow from fusion to output
    \path[arrow] (fusion.east) -- (output.west);

    % ADVANTAGE AND LIMITATION BOXES
    \node[advantage_box] (advantage) at (9, -1.5) {
        \textbf{Advantage:} \\
        AIS-augmented labels are more \\
        precise than radar-only estimates
    };

    \node[limitation_box] (limitation) at (9, -2.8) {
        \textbf{Limitation:} \\
        AIS unavailable for small craft, \\
        debris, kayaks, and uncooperative vessels
    };

\end{tikzpicture}
\caption{Triple-sensor label generation using radar, camera, and AIS. Radar provides range and bearing for projection, AIS provides vessel dimensions and class identity for bounding box sizing, and the camera provides the image substrate for annotation. AIS augmentation improves label precision but is limited to cooperative transponder-equipped vessels.}
\label{fig:ais_integration}
\end{figure}
