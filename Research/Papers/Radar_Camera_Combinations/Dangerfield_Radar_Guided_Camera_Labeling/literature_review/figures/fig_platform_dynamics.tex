\centering
\begin{tikzpicture}[scale=1.2, remember picture]

    % Define colors per user specifications
    \colorlet{garnet}{Garnet}
    \colorlet{atlantic}{Atlantic}
    \colorlet{congaree}{Congaree}
    \definecolor{darkgray}{RGB}{92,92,92}
    \definecolor{lightgray}{RGB}{199,199,199}
    % Sandstorm not in preamble - keeping local definition
  \definecolor{sandstorm}{RGB}{255, 242, 227}

    % ========== MAIN FIGURE: VESSEL DYNAMICS ==========
    \begin{scope}[shift={(0,0)}]

        % Draw wave surface
        \draw [darkgray, line width=0.5pt]
            plot [smooth, domain=-3.5:3.5, samples=100]
            (\x, {0.4*sin(2*\x r) + 0.2});

        % Fill area below waves with light color
        \fill [lightgray, opacity=0.15]
            (-4, -2) rectangle (4, 0.2);

        % ===== UPRIGHT VESSEL (DASHED) =====
        % Waterline at origin
        \draw [lightgray, dashed, line width=0.8pt] (-1.5, 0) -- (1.5, 0);

        % Hull outline (upright, dashed)
        \draw [congaree, dashed, line width=1pt]
            (-1.3, 0) -- (-1.2, -1.8) -- (1.2, -1.8) -- (1.3, 0);

        % Mast (upright, dashed)
        \draw [darkgray, dashed, line width=1.2pt] (0, 0) -- (0, 2.5);

        % Bridge structure (upright, dashed)
        \draw [lightgray, dashed, line width=0.8pt]
            (-0.6, 1.2) rectangle (0.6, 1.5);

        % Radar antenna at mast top (upright, dashed)
        \draw [garnet, dashed, line width=1.5pt]
            (0, 2.5) circle (0.15);
        \draw [garnet, dashed, line width=0.8pt]
            (-0.3, 2.5) -- (0.3, 2.5);

        % Camera at bridge (upright, dashed)
        \draw [atlantic, dashed, line width=1.5pt]
            (0, 1.35) circle (0.12);

        % ===== ROLLED VESSEL (SOLID) =====
        % Roll angle: 10 degrees
        \pgfmathsetmacro{\rollangle}{10}

        % Calculate lever arm displacements
        % h_r = 2.5 units (30m equivalent)
        % h_c = 1.35 units (20m equivalent)
        % displacement = h * sin(roll)
        \pgfmathsetmacro{\radardisp}{2.5 * sin(\rollangle)}
        \pgfmathsetmacro{\cameradisp}{1.35 * sin(\rollangle)}
        \pgfmathsetmacro{\diffdisp}{\radardisp - \cameradisp}

        % Tilted vessel center line
        \draw [congaree, solid, line width=1.2pt]
            ({-1.3*cos(\rollangle)}, {1.3*sin(\rollangle)}) --
            ({1.3*cos(\rollangle)}, {-1.3*sin(\rollangle)});

        % Hull outline (rolled)
        \path (-1.3, 0) -- (-1.2, -1.8) -- (1.2, -1.8) -- (1.3, 0);
        \begin{scope}[shift={(\radardisp, 0)}, rotate=\rollangle]
            \draw [congaree, solid, line width=1.2pt]
                (-1.3, 0) -- (-1.2, -1.8) -- (1.2, -1.8) -- (1.3, 0);

            % Mast (rolled)
            \draw [darkgray, solid, line width=1.2pt] (0, 0) -- (0, 2.5);

            % Bridge structure (rolled)
            \draw [congaree, solid, line width=0.8pt]
                (-0.6, 1.2) rectangle (0.6, 1.5);

            % Radar antenna at mast top (rolled, solid)
            \draw [garnet, solid, line width=1.5pt]
                (0, 2.5) circle (0.15);
            \draw [garnet, solid, line width=0.8pt]
                (-0.3, 2.5) -- (0.3, 2.5);
            \coordinate (radar_rolled) at (0, 2.5);

            % Camera at bridge (rolled)
            \draw [atlantic, solid, line width=1.5pt]
                (0, 1.35) circle (0.12);
            \coordinate (camera_rolled) at (0, 1.35);
        \end{scope}

        % ===== DISPLACEMENT ANNOTATIONS =====
        % Upright positions
        \coordinate (radar_upright) at (0, 2.5);
        \coordinate (camera_upright) at (0, 1.35);

        % Horizontal displacement vectors
        % Radar displacement
        \draw [garnet, ->, line width=1.2pt]
            (\radardisp, 2.5) -- (\radardisp + 0.15, 2.5);
        \draw [garnet, dashed, line width=0.8pt]
            (0, 2.5) -- (\radardisp, 2.5);

        % Camera displacement
        \draw [atlantic, ->, line width=1.2pt]
            (\cameradisp, 1.35) -- (\cameradisp + 0.1, 1.35);
        \draw [atlantic, dashed, line width=0.8pt]
            (0, 1.35) -- (\cameradisp, 1.35);

        % Differential displacement bracket
        \draw [line width=0.8pt, darkgray]
            (\radardisp, 2.3) -- (\radardisp, 2.65);
        \draw [line width=0.8pt, darkgray]
            (\cameradisp, 1.2) -- (\cameradisp, 1.5);
        \draw [line width=0.8pt, darkgray, <->, shorten >=2pt, shorten <=2pt]
            (\cameradisp - 0.05, 1.5) -- (\radardisp - 0.05, 2.3);

        % Annotations
        \node [garnet, font=\small, anchor=south] at (\radardisp + 0.4, 2.7)
            {$h_r \sin(\theta)$};
        \node [atlantic, font=\small, anchor=south] at (\cameradisp + 0.3, 1.6)
            {$h_c \sin(\theta)$};
        \node [darkgray, font=\small, anchor=west] at (\radardisp + 0.2, 1.9)
            {$\Delta = (h_r - h_c)\sin(\theta)$};
        \node [darkgray, font=\small, anchor=south] at (\cameradisp + 0.15, 0.9)
            {$\approx 0.9\text{ m}$};

        % Roll angle annotation
        \draw [darkgray, line width=0.5pt, ->] (0.4, 0) arc (0:\rollangle:0.4);
        \node [darkgray, font=\small, anchor=west] at (0.5, 0.3) {$\theta \approx 10^{\circ}$};

        % Height labels
        \draw [darkgray, line width=0.5pt] (-1.8, 0) -- (-1.7, 0);
        \draw [darkgray, line width=0.5pt] (-1.75, 0) -- (-1.75, 2.5);
        \draw [darkgray, line width=0.5pt] (-1.8, 2.5) -- (-1.7, 2.5);
        \node [darkgray, font=\footnotesize, anchor=east] at (-1.85, 1.25) {$h_r \approx 30\text{ m}$};

        \draw [darkgray, line width=0.5pt] (-1.8, 0) -- (-1.7, 0);
        \draw [darkgray, line width=0.5pt] (-1.75, 0) -- (-1.75, 1.35);
        \draw [darkgray, line width=0.5pt] (-1.8, 1.35) -- (-1.7, 1.35);
        \node [darkgray, font=\footnotesize, anchor=east] at (-1.85, 0.65) {$h_c \approx 20\text{ m}$};

    \end{scope}

    % ========== DEGREES OF FREEDOM DIAGRAM (INSET) ==========
    \begin{scope}[shift={(5.5, 1)}, scale=0.7]

        \draw [darkgray, line width=0.8pt] (-0.8, -0.8) rectangle (0.8, 0.8);
        \node [darkgray, font=\small, anchor=south] at (0, 0.95) {6 DOF};

        % Draw vessel outline
        \draw [congaree, line width=0.8pt]
            (-0.5, -0.2) -- (-0.4, -0.6) -- (0.4, -0.6) -- (0.5, -0.2);

        % Surge (forward/back)
        \draw [darkgray, ->, line width=0.8pt] (-0.7, 0.6) -- (0.7, 0.6);
        \node [darkgray, font=\footnotesize] at (0, 0.75) {surge};

        % Sway (left/right)
        \draw [darkgray, ->, line width=0.8pt] (0.7, 0.2) -- (0.7, -0.4);
        \node [darkgray, font=\footnotesize, anchor=west] at (0.8, -0.1) {sway};

        % Heave (up/down)
        \draw [darkgray, ->, line width=0.8pt] (-0.7, 0.2) -- (-0.7, -0.4);
        \node [darkgray, font=\footnotesize, anchor=east] at (-0.8, -0.1) {heave};

        % Roll (rotate about length axis)
        \draw [darkgray, ->, line width=0.8pt]
            (0.2, -0.5) arc (0:90:0.2);
        \node [darkgray, font=\footnotesize] at (0.35, -0.25) {roll};

        % Pitch (rotate about width axis)
        \draw [darkgray, ->, line width=0.8pt]
            (-0.2, 0.5) arc (180:270:0.15);
        \node [darkgray, font=\footnotesize] at (-0.4, 0.4) {pitch};

        % Yaw (rotate about vertical axis)
        \draw [darkgray, ->, line width=0.8pt]
            (0.15, 0.2) arc (45:135:0.15);
        \node [darkgray, font=\footnotesize] at (0, 0.45) {yaw};

    \end{scope}

    % ========== CALIBRATION INVALIDATION INSET ==========
    \begin{scope}[shift={(5.5, -1.5)}, scale=0.65]

        \draw [garnet, line width=1.2pt, fill=sandstorm, fill opacity=0.2]
            (-1.2, -0.8) rectangle (1.2, 0.8);

        \node [garnet, font=\small, anchor=north, align=center] at (0, 0.6)
            {Static Calibration};
        \node [garnet, font=\small, anchor=north, align=center] at (0, 0.2)
            {Invalidated by};
        \node [garnet, font=\small, anchor=north, align=center] at (0, -0.2)
            {Dynamic Motion};
        \node [garnet, font=\footnotesize, anchor=north, align=center] at (0, -0.6)
            {Requires real-time IMU compensation};

    \end{scope}

\end{tikzpicture}

\caption{Platform dynamics and lever-arm effects in maritime radar-camera calibration. Vessel roll causes differential lateral displacement between the radar antenna (mounted at mast height $h_r$) and the camera (at bridge height $h_c$), introducing time-varying extrinsic calibration errors that must be compensated using inertial measurements.}
