\end{center}
\vspace{0.5cm}

\newcommand{\cfarGrid}[3]{% #1=x-offset, #2=title, #3=style key
\begin{scope}[shift={(#1,0)}]
  \node[font=\bfseries] at (2.5,5.8) {#2};
  % Draw 5x5 grid
  \foreach \r in {0,...,4}{
    \foreach \c in {0,...,4}{
      \pgfmathtruncatemacro{\iscut}{(\r==2 && \c==2) ? 1 : 0}
      \pgfmathtruncatemacro{\isguard}{(abs(\r-2)<=1 && abs(\c-2)<=1 && \iscut==0) ? 1 : 0}
      % leading = top rows (r<2 after guard exclusion)
      \pgfmathtruncatemacro{\isleading}{(\iscut==0 && \isguard==0 && \r<2) ? 1 : 0}
      \pgfmathtruncatemacro{\istrailing}{(\iscut==0 && \isguard==0 && \r>2) ? 1 : 0}
      %
      \ifnum\iscut=1
        \fill[garnet] (\c,\r) rectangle ++(1,1);
      \else\ifnum\isguard=1
        \fill[gray50] (\c,\r) rectangle ++(1,1);
      \else
        % Style depends on variant
        \ifnum#3=0 % CA: all reference equal
          \fill[atlantic!60] (\c,\r) rectangle ++(1,1);
        \fi
        \ifnum#3=1 % GO: leading bold, trailing dim
          \ifnum\isleading=1
            \fill[atlantic!80] (\c,\r) rectangle ++(1,1);
          \else
            \fill[atlantic!25] (\c,\r) rectangle ++(1,1);
          \fi
        \fi
        \ifnum#3=2 % SO: trailing bold, leading dim
          \ifnum\istrailing=1
            \fill[atlantic!80] (\c,\r) rectangle ++(1,1);
          \else
            \fill[atlantic!25] (\c,\r) rectangle ++(1,1);
          \fi
        \fi
      \fi\fi
      \draw[gray70] (\c,\r) rectangle ++(1,1);
    }
  }
  % Annotations below
  \ifnum#3=0
    \node[font=\small,text=gray70] at (2.5,-0.5) {All reference cells averaged};
  \fi
  \ifnum#3=1
    \node[font=\small,text=gray70] at (2.5,-0.5) {Max(leading, trailing) selected};
  \fi
  \ifnum#3=2
    \node[font=\small,text=gray70] at (2.5,-0.5) {Min(leading, trailing) selected};
  \fi
\end{scope}
}

\begin{center}
\begin{tikzpicture}[scale=0.7]
  \cfarGrid{0}{CA-CFAR}{0}
  \cfarGrid{6.5}{GO-CFAR}{1}
  \cfarGrid{13}{SO-CFAR}{2}

  % Legend
  \fill[garnet] (0,-1.8) rectangle ++(0.5,0.5);
  \node[right,font=\small] at (0.6,-1.55) {Cell Under Test};
  \fill[gray50] (5,-1.8) rectangle ++(0.5,0.5);
  \node[right,font=\small] at (5.6,-1.55) {Guard Region};
  \fill[atlantic!60] (10,-1.8) rectangle ++(0.5,0.5);
  \node[right,font=\small] at (10.6,-1.55) {Reference Cells};
\end{tikzpicture}
\end{center}
