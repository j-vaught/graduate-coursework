\question{Problem 1: Vector Operations (10 pts)}\label{prob:1}

Create two vectors $\mathbf{v}_1 = [-4,\; 3,\; 5]^T$ and $\mathbf{v}_2 = [1,\; 0,\; 7]^T$. Find the scalar (inner) product and outer product. Show that $\mathbf{v}_1^T \mathbf{v}_2 = \mathrm{Tr}(\mathbf{v}_1 \mathbf{v}_2^T)$.

\begin{codebox}
\begin{lstlisting}
v_1 = [-4; 3; 5];
v_2 = [1; 0; 7];

scalar_product = v_1' * v_2
outer_product = v_1 * v_2'
checkEqual(v_1' * v_2, trace(v_1 * v_2'))
\end{lstlisting}
\end{codebox}

\begin{outputbox}
\begin{lstlisting}[style=output]
scalar_product =
    31

outer_product =
    -4     0   -28
     3     0    21
     5     0    35

The two inputs are equal
\end{lstlisting}
\end{outputbox}

\begin{resultsbox}
\textbf{Scalar (inner) product:}
\[
\mathbf{v}_1^T \mathbf{v}_2 = (-4)(1) + (3)(0) + (5)(7) = 31
\]

\textbf{Outer product:}
\[
\mathbf{v}_1 \mathbf{v}_2^T = \begin{pmatrix} -4 \\ 3 \\ 5 \end{pmatrix} \begin{pmatrix} 1 & 0 & 7 \end{pmatrix} = \begin{pmatrix} -4 & 0 & -28 \\ 3 & 0 & 21 \\ 5 & 0 & 35 \end{pmatrix}
\]

\textbf{Trace identity:}
$\mathrm{Tr}(\mathbf{v}_1 \mathbf{v}_2^T) = -4 + 0 + 35 = 31 = \mathbf{v}_1^T \mathbf{v}_2$ \quad \checkmark
\end{resultsbox}

\question{Problem 2: Matrix Properties (15 pts)}\label{prob:2}

Define:
\[
A = \begin{pmatrix} -3 & 5 & 7 \\ 1 & 2 & 4 \\ 6 & 8 & -9 \end{pmatrix}, \qquad
B = \begin{pmatrix} 8 & 1 & 6 \\ -3 & 5 & 7 \\ 4 & 9 & -2 \end{pmatrix}
\]

% --- 2.1 ---
\questionpart{Problem 2.1: Transpose Identities}\label{prob:2.1}

Show that $(A^T)^T = A$, \; $(AB)^T = B^T A^T$, \; and $(A+B)^T = A^T + B^T$.

\begin{codebox}
\begin{lstlisting}
A = [-3, 5, 7; 1, 2, 4; 6, 8, -9];
B = [8, 1, 6; -3, 5, 7; 4, 9, -2];

checkEqual((A')', A)
checkEqual((A * B)', B' * A')
checkEqual((A + B)', A' + B')
\end{lstlisting}
\end{codebox}

\begin{outputbox}
\begin{lstlisting}[style=output]
The two inputs are equal
The two inputs are equal
The two inputs are equal
\end{lstlisting}
\end{outputbox}

\begin{resultsbox}
All three transpose identities verified:
\begin{itemize}
    \item $(A^T)^T = A$ \quad \checkmark
    \item $(AB)^T = B^T A^T$ \quad \checkmark
    \item $(A+B)^T = A^T + B^T$ \quad \checkmark
\end{itemize}
\end{resultsbox}

\question{Problem 3: Matrix Inversion Lemma --- Example 1.1 (25 pts)}\label{prob:3}

Verify the matrix inversion lemma (Equation 1.45) using Example 1.1 (p.\ 12). Given:
\[
A = \begin{pmatrix} -2 & 1 & 2 \\ -4 & 3 & 2 \\ -5 & 1 & 5 \end{pmatrix}, \quad
A' = \begin{pmatrix} -2 & 1 & 2 \\ -4 & 3 & 2 \\ -5 & 1 & 6 \end{pmatrix}
\]
where $A' = A + BDC$ with $B = [0;\; 0;\; 1]$, $C = [0\;\; 0\;\; 1]$, $D = 1$.

The matrix inversion lemma states:
\[
(A + BD^{-1}C)^{-1} = A^{-1} - A^{-1}B(D + CA^{-1}B)^{-1}CA^{-1}
\]

\begin{codebox}
\begin{lstlisting}
A = [-2 1 2; -4 3 2; -5 1 5];
A_inv = inv(A);
A_acute = [-2 1 2; -4 3 2; -5 1 6];
B = [0; 0; 1];
C = [0 0 1];
D = 1;

A_acute_inv_lemma = A_inv - A_inv*B*inv(D + C*A_inv*B)*C*A_inv
A_acute_inv_matlab = inv(A_acute)
checkEqual(A_acute_inv_lemma, A_acute_inv_matlab)
\end{lstlisting}
\end{codebox}

\begin{outputbox}
\begin{lstlisting}[style=output]
A_acute_inv_lemma =
    4.0000   -1.0000   -1.0000
    3.5000   -0.5000   -1.0000
    2.7500   -0.7500   -0.5000

A_acute_inv_matlab =
    4.0000   -1.0000   -1.0000
    3.5000   -0.5000   -1.0000
    2.7500   -0.7500   -0.5000

The two inputs are equal
\end{lstlisting}
\end{outputbox}

\begin{resultsbox}
The matrix inversion lemma yields:
\[
(A')^{-1} = A^{-1} - A^{-1}B(D + CA^{-1}B)^{-1}CA^{-1} = \begin{pmatrix} 4 & -1 & -1 \\ 3.5 & -0.5 & -1 \\ 2.75 & -0.75 & -0.5 \end{pmatrix}
\]
This matches MATLAB's \texttt{inv(A\_acute)} exactly. \quad \checkmark
\end{resultsbox}
