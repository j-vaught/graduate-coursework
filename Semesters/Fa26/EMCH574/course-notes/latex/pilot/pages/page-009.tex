\section*{Complex-exponential form}

The solution can also be written as
\begin{equation}
    u(t)=C_1e^{i\omega_n t}+C_2e^{-i\omega_n t}.
    \tag{1}
\end{equation}
The initial-value conditions are
\begin{equation}
\begin{aligned}
    u(0) &= u_0,\\
    \dot{u}(0) &= \dot{u}_0.
\end{aligned}
    \tag{2}
\end{equation}
Differentiating Eq.~(1) gives
\begin{equation}
    \dot{u}(t)=i\omega_n C_1e^{i\omega_n t}-i\omega_n C_2e^{-i\omega_n t}.
    \tag{3}
\end{equation}
Thus,
\begin{equation}
\begin{aligned}
    u(0) &= C_1+C_2,\\
    \dot{u}(0) &= i\omega_n C_1-i\omega_n C_2.
\end{aligned}
    \tag{4}
\end{equation}
Using Eq.~(4) with the initial conditions gives
\begin{equation}
\begin{aligned}
    C_1+C_2 &= u_0,\\
    i\omega_n C_1-i\omega_n C_2 &= \dot{u}_0.
\end{aligned}
    \tag{5}
\end{equation}
Adding and subtracting these equations yields
\begin{equation}
\begin{aligned}
    2i\omega_n C_1 &= i\omega_n u_0+\dot{u}_0,\\
    -2i\omega_n C_2 &= -i\omega_n u_0+\dot{u}_0,
\end{aligned}
    \tag{6}
\end{equation}
and therefore
\begin{equation}
\begin{aligned}
    C_1 &= \frac{u_0}{2}+\frac{\dot{u}_0}{2i\omega_n},\\
    C_2 &= \frac{u_0}{2}-\frac{\dot{u}_0}{2i\omega_n}.
\end{aligned}
    \tag{7}
\end{equation}
Substitution into Eq.~(1) gives
\begin{align*}
u(t) &= \left(\frac{u_0}{2}+\frac{\dot{u}_0}{2i\omega_n}\right)e^{i\omega_n t}
 +\left(\frac{u_0}{2}-\frac{\dot{u}_0}{2i\omega_n}\right)e^{-i\omega_n t}\\
&=u_0\frac{e^{i\omega_n t}+e^{-i\omega_n t}}{2}
 +\frac{\dot{u}_0}{\omega_n}\frac{e^{i\omega_n t}-e^{-i\omega_n t}}{2i}\\
&=u_0\cos\omega_n t+\frac{\dot{u}_0}{\omega_n}\sin\omega_n t,
\end{align*}
as before.
