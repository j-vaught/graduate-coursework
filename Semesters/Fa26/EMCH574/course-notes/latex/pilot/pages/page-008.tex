\section*{Oscillatory motion}

\PilotFigure{p008-oscillatory-motion}{Oscillatory motion with amplitude and phase.}

The displacement may be written in either of the equivalent forms
\begin{equation}
    u(t) = C\sin(\omega_n t+\varphi).
    \tag{13a}
\end{equation}
\begin{equation}
    u(t) = A\cos\omega_n t+B\sin\omega_n t.
    \tag{13b}
\end{equation}
The constants $A$, $B$ or $C$, $\varphi$ must be determined from the initial conditions.

\subsection*{Initial-value problem}

Assume that the initial displacement, $u_0$, and the initial velocity, $\dot{u}_0$, are known. Using Eq.~(13b), we write
\begin{equation}
\begin{aligned}
    u_0 = u(0) &= \left.\left(A\cos\omega_n t+B\sin\omega_n t\right)\right|_{t=0}=A,\\
    \dot{u}_0 = \dot{u}(0) &= \left.\left(-\omega_n A\sin\omega_n t+\omega_n B\cos\omega_n t\right)\right|_{t=0}=\omega_n B.
\end{aligned}
    \tag{26}
\end{equation}
Solving Eq.~(26) for $A$ and $B$ yields
\begin{equation}
\begin{aligned}
    A &= u_0,\\
    B &= \frac{\dot{u}_0}{\omega_n}.
\end{aligned}
    \tag{27}
\end{equation}
Substitution of Eq.~(27) into Eq.~(13b) gives the general solution of undamped free vibrations with initial displacement, $u_0$, and initial velocity, $\dot{u}_0$, in the form
\begin{equation}
    u(t)=u_0\cos\omega_n t+\frac{\dot{u}_0}{\omega_n}\sin\omega_n t.
    \tag{28}
\end{equation}
