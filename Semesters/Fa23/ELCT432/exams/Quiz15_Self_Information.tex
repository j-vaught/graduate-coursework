\section*{Answers and Explanations}

\begin{enumerate}
    \item \textbf{Amount of Information for Observing “1” at the Source:}
    \[ I(1) = -\log_2 P(X=1) = -\log_2 \left( \frac{3}{4} \right) \approx 0.415 \text{ bits} \]
    This equation calculates the information content for the event \( X=1 \) using the formula for information content in bits.

    \item \textbf{Amount of Information for Observing “0” at the Source:}
    \[ I(0) = -\log_2 P(X=0) = -\log_2 \left( \frac{1}{4} \right) = 2.000 \text{ bits} \]
    This represents the information content for the event \( X=0 \).

    \item \textbf{Average Information at the Source (Entropy of X):}
    \[ H(X) = -\sum_{x \in \{0,1\}} P(X=x) \log_2 P(X=x) \approx 0.811 \text{ bits} \]
    Entropy \( H(X) \) is the average uncertainty in the random variable \( X \).

    \item \textbf{Probability that the Output is “1”:}
    \[ P(Y=1) = P(X=1)(1 - \text{flip probability}) + P(X=0)(\text{flip probability}) = 0.625 \]

    \item \textbf{Probability that the Output is “0”:}
    \[ P(Y=0) = P(X=0)(1 - \text{flip probability}) + P(X=1)(\text{flip probability}) = 0.375 \]

    \item \textbf{Average Information at the Output (Entropy of Y):}
    \[ H(Y) = -\sum_{y \in \{0,1\}} P(Y=y) \log_2 P(Y=y) \approx 0.954 \text{ bits} \]

    \item \textbf{Probability of Transmitting 0 and Receiving 0:}
    \[ P(0 \text{ transmitted, } 0 \text{ received}) = P(X=0)(1 - \text{flip probability}) = 0.1875 \]

    \item \textbf{Probability of Transmitting 1 and Receiving 0:}
    \[ P(1 \text{ transmitted, } 0 \text{ received}) = P(X=1)(\text{flip probability}) = 0.1875 \]

    \item \textbf{Probability of Transmitting 0 and Receiving 1:}
    \[ P(0 \text{ transmitted, } 1 \text{ received}) = P(X=0)(\text{flip probability}) = 0.0625 \]

    \item \textbf{Probability of Transmitting 1 and Receiving 1:}
    \[ P(1 \text{ transmitted, } 1 \text{ received}) = P(X=1)(1 - \text{flip probability}) = 0.5625 \]

    \item \textbf{Joint Entropy H(X, Y):}
    \[ H(X, Y) = -\sum_{x \in \{0,1\}} \sum_{y \in \{0,1\}} P(X=x, Y=y) \log_2 P(X=x, Y=y) \approx 1.623 \text{ bits} \]
\end{enumerate}
