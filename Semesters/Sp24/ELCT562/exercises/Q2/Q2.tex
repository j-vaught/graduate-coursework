\section{Problem Statement}
Assume that the transmit power is 1 Watt.

\subsection{Calculation of the Transmit Power in dBm}
The transmit power in dBm, when the transmit power is 1 Watt, is calculated as follows:
\begin{align*}
\text{Transmit Power (dBm)} &= 10 \log_{10}(\text{Transmit Power (Watts)} \times 1000) \\
&= 10 \log_{10}(1 \times 1000) \\
&= 30 \, \text{dBm}
\end{align*}

\subsection{Calculation of the Received Signal Power in dBm using Friis' Path Loss Model}
\subsubsection{For 1500 MHz at 100 meters}
\begin{itemize}
\item Frequency (Hz): $f = 1500 \times 10^6$
\item Distance (meters): $r = 100$
\item Antenna Gains: $G_t = G_r = 10$ dBi
\item Path Loss (PL):
\begin{align*}
PL &= 20 \log_{10}(100) + 20 \log_{10}(1500 \times 10^6) - 147.55 - 10 - 10 \\
&\approx 55.97 \, \text{dB}
\end{align*}
\item Received Power (dBm):
\begin{align*}
\text{Received Power (dBm)} &= 30 \, \text{dBm} - 55.97 \, \text{dB} \\
&\approx -25.97 \, \text{dBm}
\end{align*}
\end{itemize}

\subsubsection{For 60 GHz at 100 meters}
\begin{itemize}
\item Frequency (Hz): $f = 60 \times 10^9$
\item Distance (meters): $r = 100$
\item Antenna Gains: $G_t = G_r = 10$ dBi
\item Path Loss (PL):
\begin{align*}
PL &= 20 \log_{10}(100) + 20 \log_{10}(60 \times 10^9) - 147.55 - 10 - 10 \\
&\approx 88.01 \, \text{dB}
\end{align*}
\item Received Power (dBm):
\begin{align*}
\text{Received Power (dBm)} &= 30 \, \text{dBm} - 88.01 \, \text{dB} \\
&\approx -58.01 \, \text{dBm}
\end{align*}
\end{itemize}
