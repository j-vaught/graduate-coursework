\begin{frame}{Why Depth Matters}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Knowing 3D from 2D enables:}
\begin{itemize}
  \item \textcolor{atlantic}{\textbf{Autonomous driving}} --- obstacle distance, lane geometry
  \item \textcolor{congaree}{\textbf{Robotics}} --- grasping, navigation, collision avoidance
  \item \textcolor{horseshoe}{\textbf{AR / VR}} --- object occlusion, scene placement
  \item \textcolor{garnet}{\textbf{3D reconstruction}} --- single-image scene modeling
\end{itemize}

\vspace{0.3cm}
All require \textit{metric} depth --- real-world distances in meters, not just relative ordering.

\column{0.48\textwidth}
\textbf{My use case: marine radar labeling}
