local CFG = {
    -- Lens geometry
    lens1_x  = 3.0,      -- center x of lens 1
    lens2_x  = 5.5,      -- center x of lens 2
    lens1_R  = 6.0,      -- radius of curvature (larger = weaker, rays diverge more)
    lens2_R  = 8.0,
    lens1_h  = 1.8,      -- half-aperture height (bigger than sensor)
    lens2_h  = 1.6,
    n_glass  = 1.52,     -- refractive index

    -- Sensor
    sensor_x = 6.8,      -- closer to lenses = rays diverge after exit
    sensor_h = 0.8,      -- smaller sensor, lenses collect more light

    -- Ray source: parallel beam
    n_rays      = 40,       -- number of parallel rays
    ray_y_min   = -0.7,     -- match sensor height
    ray_y_max   =  0.7,     -- highest ray y

    -- Ray source: fan from a point
    fan_on      = false,    -- enable/disable fan
    fan_x       = -0.3,     -- source point x
    fan_y       =  0.9,     -- source point y
    fan_n       = 30,       -- number of fan rays
    fan_ang_min = -20,      -- min angle (degrees, 0 = horizontal right)
    fan_ang_max =  5,       -- max angle

    -- Appearance
    ray_width      = 0.2,   -- line width in pt
    parallel_color = "garnet",
    fan_color      = "atlantic",
    parallel_opacity = 0.5,
    fan_opacity      = 0.35,

    -- Scene bounds
    xmin = -0.5, xmax = 8.8,
    ymin = -2.0, ymax = 2.0,
}

-- ════════════════════════════════════════════════════════════
-- ENGINE — no need to edit below here
-- ════════════════════════════════════════════════════════════
local C = CFG

local function sag(R, h) return R - math.sqrt(math.max(R*R - h*h, 0)) end

local function isect(ox, oy, dx, dy, cx, cy, r, pick)
    local ex, ey = ox - cx, oy - cy
    local a = dx*dx + dy*dy
    local b = 2*(ex*dx + ey*dy)
    local c = ex*ex + ey*ey - r*r
    local disc = b*b - 4*a*c
    if disc < 0 then return nil end
    local sq = math.sqrt(disc)
    local t1 = (-b - sq)/(2*a)
    local t2 = (-b + sq)/(2*a)
    local t
    if pick == "far" then
        t = (t2 > 1e-4) and t2 or ((t1 > 1e-4) and t1 or nil)
    else
        t = (t1 > 1e-4) and t1 or ((t2 > 1e-4) and t2 or nil)
    end
    if not t then return nil end
    local hx, hy = ox + t*dx, oy + t*dy
    return t, hx, hy, (hx-cx)/r, (hy-cy)/r
end

local function refract(dx, dy, nx, ny, n1, n2)
    local d = dx*nx + dy*ny
    if d > 0 then nx, ny, d = -nx, -ny, -d end
    local r = n1/n2
    local ci = -d
    local s2t = r*r*(1 - ci*ci)
    if s2t > 1 then return nil end
    local ct = math.sqrt(1 - s2t)
    local rx = r*dx + (r*ci - ct)*nx
    local ry = r*dy + (r*ci - ct)*ny
    local l = math.sqrt(rx*rx + ry*ry)
    return rx/l, ry/l
end

local function lens_circles(lx, R, h)
    local d = math.sqrt(math.max(R*R - h*h, 0))
    return lx + d, lx - d
end

local function draw_lens(lx, R, h)
    local cxL, cxR = lens_circles(lx, R, h)
    local N = 60
    local left, right = {}, {}
    for i = 0, N do
        local y = -h + 2*h*(i/N)
        local q = math.sqrt(math.max(R*R - y*y, 0))
        table.insert(left,  {cxL - q, y})
        table.insert(right, {cxR + q, y})
    end
    local p = "\\fill[atlantic!10, opacity=0.85] "
    for i, v in ipairs(left) do
        p = p .. (i==1 and "" or " -- ") .. string.format("(%.4f,%.4f)", v[1], v[2])
    end
    for i = #right, 1, -1 do
        p = p .. string.format(" -- (%.4f,%.4f)", right[i][1], right[i][2])
    end
    tex.sprint(p .. " -- cycle;")
    for _, arc in ipairs({left, right}) do
        local s = "\\draw[atlantic!60, thin] "
        for i, v in ipairs(arc) do
            s = s .. (i==1 and "" or " -- ") .. string.format("(%.4f,%.4f)", v[1], v[2])
        end
        tex.sprint(s .. ";")
    end
    tex.sprint(string.format(
        "\\draw[atlantic!60, thin] (%.4f,%.4f)--(%.4f,%.4f);",
        left[1][1], left[1][2], right[1][1], right[1][2]))
    tex.sprint(string.format(
        "\\draw[atlantic!60, thin] (%.4f,%.4f)--(%.4f,%.4f);",
        left[#left][1], left[#left][2], right[#right][1], right[#right][2]))
end

-- Trace ray through lenses. Direction-aware: if dx < 0, traverse lenses in reverse.
local function trace(ox, oy, dx, dy)
    local pts = {{ox, oy}}
    local cx, cy, cdx, cdy = ox, oy, dx, dy
    local lenses
    if dx >= 0 then
        -- Left to right: lens1 then lens2
        lenses = {
            {x=C.lens1_x, R=C.lens1_R, h=C.lens1_h},
            {x=C.lens2_x, R=C.lens2_R, h=C.lens2_h},
        }
    else
        -- Right to left: lens2 then lens1
        lenses = {
            {x=C.lens2_x, R=C.lens2_R, h=C.lens2_h},
            {x=C.lens1_x, R=C.lens1_R, h=C.lens1_h},
        }
    end
    for _, L in ipairs(lenses) do
        local cxL, cxR = lens_circles(L.x, L.R, L.h)
        -- Going right: hit left surface (near on cxL), then right surface (far on cxR)
        -- Going left:  hit right surface (near on cxR), then left surface (far on cxL)
        local surf1_cx, surf1_pick, surf2_cx, surf2_pick
        if dx >= 0 then
            surf1_cx, surf1_pick = cxL, "near"
            surf2_cx, surf2_pick = cxR, "far"
        else
            surf1_cx, surf1_pick = cxR, "near"
            surf2_cx, surf2_pick = cxL, "far"
        end
        local t, hx, hy, nx, ny = isect(cx, cy, cdx, cdy, surf1_cx, 0, L.R, surf1_pick)
        if t and math.abs(hy) <= L.h then
            table.insert(pts, {hx, hy})
            local ndx, ndy = refract(cdx, cdy, nx, ny, 1.0, C.n_glass)
            if not ndx then break end
            cdx, cdy = ndx, ndy
            cx, cy = hx, hy
            t, hx, hy, nx, ny = isect(cx, cy, cdx, cdy, surf2_cx, 0, L.R, surf2_pick)
            if t and math.abs(hy) <= L.h then
                table.insert(pts, {hx, hy})
                ndx, ndy = refract(cdx, cdy, nx, ny, C.n_glass, 1.0)
                if not ndx then break end
                cdx, cdy = ndx, ndy
                cx, cy = hx, hy
            end
        end
    end
    -- Extend to scene boundary
    if math.abs(cdx) > 1e-6 then
        local target_x = (cdx > 0) and C.xmax or C.xmin
        local ts = (target_x - cx) / cdx
        if ts > 0 then
            table.insert(pts, {target_x, cy + ts*cdy})
        end
    end
    return pts
end

local function draw_rays(pts, color, opacity, width)
    for j = 1, #pts-1 do
        tex.sprint(string.format(
            "\\draw[%s, line width=%.2fpt, opacity=%.2f] (%.4f,%.4f)--(%.4f,%.4f);",
            color, width, opacity,
            pts[j][1], pts[j][2], pts[j+1][1], pts[j+1][2]))
    end
end

-- Optical axis
tex.sprint(string.format(
    "\\draw[warmgrey, dashed, thin] (%.2f,0)--(%.2f,0);", C.xmin, C.xmax))

-- Lenses
draw_lens(C.lens1_x, C.lens1_R, C.lens1_h)
draw_lens(C.lens2_x, C.lens2_R, C.lens2_h)

-- CMOS
tex.sprint(string.format(
    "\\fill[congaree!80] (%.2f,%.2f) rectangle (%.2f,%.2f);",
    C.sensor_x, -C.sensor_h, C.sensor_x+0.12, C.sensor_h))
for i = 0, 15 do
    local y = -C.sensor_h + i*(2*C.sensor_h/16)
    tex.sprint(string.format(
        "\\draw[congaree!40, very thin] (%.2f,%.4f)--(%.2f,%.4f);",
        C.sensor_x, y, C.sensor_x+0.12, y))
end
tex.sprint(string.format(
    "\\draw[congaree, thick] (%.2f,%.2f) rectangle (%.2f,%.2f);",
    C.sensor_x, -C.sensor_h, C.sensor_x+0.12, C.sensor_h))

-- Parallel rays from CMOS going LEFT into the world
for i = 0, C.n_rays - 1 do
    local frac = (C.n_rays > 1) and (i / (C.n_rays - 1)) or 0.5
    local y0 = C.ray_y_min + (C.ray_y_max - C.ray_y_min) * frac
    local pts = trace(C.sensor_x, y0, -1, 0)
    draw_rays(pts, C.parallel_color, C.parallel_opacity, C.ray_width)
end

-- Fan rays from a world-side point source going RIGHT into the camera
if C.fan_on then
    for i = 0, C.fan_n - 1 do
        local frac = (C.fan_n > 1) and (i / (C.fan_n - 1)) or 0.5
        local ang = math.rad(C.fan_ang_min + (C.fan_ang_max - C.fan_ang_min) * frac)
        local pts = trace(C.fan_x, C.fan_y, math.cos(ang), math.sin(ang))
        draw_rays(pts, C.fan_color, C.fan_opacity, C.ray_width)
    end
    tex.sprint(string.format("\\fill[atlantic] (%.2f,%.2f) circle(0.05);", C.fan_x, C.fan_y))
end

-- Labels
tex.sprint(string.format(
    "\\node[font=\\sffamily\\footnotesize,text=atlantic,anchor=south] at (%.2f,%.2f) {Lens 1};",
    C.lens1_x, C.lens1_h+0.1))
tex.sprint(string.format(
    "\\node[font=\\sffamily\\footnotesize,text=atlantic,anchor=south] at (%.2f,%.2f) {Lens 2};",
    C.lens2_x, C.lens2_h+0.1))
tex.sprint(string.format(
    "\\node[font=\\sffamily\\footnotesize,text=congaree,anchor=south] at (%.2f,%.2f) {CMOS};",
    C.sensor_x+0.06, C.sensor_h+0.1))
tex.sprint(string.format(
    "\\node[font=\\sffamily\\scriptsize,text=warmgrey,anchor=south west] at (%.2f,%.2f) {$R_1=%.1f$ \\quad $R_2=%.1f$ \\quad $d=%.1f$};",
    C.xmin+0.1, C.ymin+0.05, C.lens1_R, C.lens2_R, C.lens2_x - C.lens1_x))
