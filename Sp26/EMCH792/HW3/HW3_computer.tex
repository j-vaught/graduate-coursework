\question{Problem B1: Exercise 2.15 --- Uniform Random Variables (20 pts)}\label{prob:b1}
Generate independent random variables uniformly distributed on $(0,1)$ for $N=50, 500, 5000$. Plot 10-bin histograms and report sample mean/standard deviation. Compare against theoretical values:
\[
\mu = 0.5, \qquad \sigma = \frac{1}{\sqrt{12}} \approx 0.288675
\]

\begin{codebox}
\begin{lstlisting}
rng(792);
N_values = [50, 500, 5000];
theory_mean = 0.5;
theory_std = 1/sqrt(12);

for i = 1:numel(N_values)
    N = N_values(i);
    u = rand(N, 1);
    sample_mean(i) = mean(u);
    sample_std(i) = std(u);
end
\end{lstlisting}
\end{codebox}

\begin{outputbox}
\begin{lstlisting}[style=output]
Theoretical mean = 0.500000, theoretical std = 0.288675
N=  50 -> sample mean = 0.466955, sample std = 0.248034
N= 500 -> sample mean = 0.497011, sample std = 0.286045
N=5000 -> sample mean = 0.508207, sample std = 0.289937
\end{lstlisting}
\end{outputbox}

\begin{figure}[H]
\centering
\includegraphics[width=0.98\textwidth]{HW3_B1_histograms.png}
\caption{Exercise 2.15 histograms (10 bins) for $N=50, 500, 5000$.}

\begin{resultsbox}
As $N$ increases, the sample statistics converge toward the theoretical values and the histogram shape becomes smoother and more representative of the uniform distribution.
\end{resultsbox}

\question{Problem B2: Noise Covariance Matrix Calculation (20 pts)}\label{prob:b2}
Given
\[
\dot x(t) = A x(t) + B_u u(t) + B_w w(t),
\]
\[
A = \begin{bmatrix}-1 & 1\\-2 & -2\end{bmatrix},\quad
B_u = \begin{bmatrix}0\\-1\end{bmatrix},\quad
B_w = \begin{bmatrix}1\\2\end{bmatrix},\quad S_w = 0.11.
\]
with $\Delta t_1 = 0.5$ and $\Delta t_2 = 0.01$.

Define the continuous covariance intensity:
\[
Q_c = B_w S_w B_w^T = \begin{bmatrix}0.11 & 0.22\\0.22 & 0.44\end{bmatrix}.
\]

\questionpart{B2.1: Exact and Approximate $\Sigma_{\tilde w}$}\label{prob:b2a}
Using Van Loan's block matrix:
\[
Z = \begin{bmatrix}-A & Q_c\\0 & A^T\end{bmatrix}\Delta t,
\]
\[
\exp(Z)=\begin{bmatrix}E_{11} & E_{12}\\0 & E_{22}\end{bmatrix},\quad
A_d = E_{11}^{-1},\quad \Sigma_{\tilde w}^{\text{exact}} = A_d E_{12},
\]
and first-order approximation $\Sigma_{\tilde w}^{\text{approx}} = Q_c \Delta t$.

\begin{outputbox}
\begin{lstlisting}[style=output]
dt = 0.500
Sigma_w_exact =
    0.0590    0.0507
    0.0507    0.0591
Sigma_w_approx =
    0.0550    0.1100
    0.1100    0.2200
||difference||_F = 0.1814999427

dt = 0.010
Sigma_w_exact =
    0.0011    0.0022
    0.0022    0.0043
Sigma_w_approx =
    0.0011    0.0022
    0.0022    0.0044
||difference||_F = 0.0001341646
\end{lstlisting}
\end{outputbox}

\begin{resultsbox}
The approximation error is large at $\Delta t=0.5$ but very small at $\Delta t=0.01$, confirming that $Q_c\Delta t$ is only accurate for sufficiently small sampling periods.
\end{resultsbox}

\questionpart{B2.3: Continuous/Discrete Steady-State Covariances}\label{prob:b2c}
Continuous Lyapunov equation:
\[
A\Sigma_{x,ss}^{(c)} + \Sigma_{x,ss}^{(c)}A^T + Q_c = 0.
\]
Discrete Lyapunov equation:
\[
\Sigma_{x,ss}^{(d)} = A_d\Sigma_{x,ss}^{(d)}A_d^T + \Sigma_{\tilde w}^{\text{exact}}.
\]

\begin{outputbox}
\begin{lstlisting}[style=output]
Sigma_x_ss_cont =
    0.0917    0.0367
    0.0367    0.0733

dt = 0.500
Sigma_x_ss_disc =
    0.0917    0.0367
    0.0367    0.0733
||disc - cont||_F = 3.6717175287e-17

dt = 0.010
Sigma_x_ss_disc =
    0.0917    0.0367
    0.0367    0.0733
||disc - cont||_F = 3.3001717888e-16
\end{lstlisting}
\end{outputbox}

\begin{resultsbox}
When the exact discrete process covariance is used, continuous and discrete steady-state covariance predictions agree (numerical differences only at machine precision).
\end{resultsbox}
