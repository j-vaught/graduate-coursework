\question{Exercise 2.5 (Page 75): Exponentially Distributed Random Variable}\label{ex:2.5}
Given
\[
f_X(x)=
\begin{cases}
\alpha e^{-\alpha x}, & x>0,\\
0, & x\le 0,
\end{cases}
\qquad \alpha > 0.
\]

\begin{stepbox}
\textbf{(a) Probability distribution function (CDF)}
\[
F_X(x)=P(X\le x)=
\begin{cases}
0, & x\le 0,\\
\int_{0}^{x}\alpha e^{-\alpha z}\,dz = 1-e^{-\alpha x}, & x>0.
\end{cases}
\]

\textbf{(b) Mean}
\[
\mu_X = E[X] = \int_0^{\infty} x\alpha e^{-\alpha x}dx = \frac{1}{\alpha}.
\]

\textbf{(c) Second moment}
\[
E[X^2] = \int_0^{\infty} x^2\alpha e^{-\alpha x}dx = \frac{2}{\alpha^2}.
\]

\textbf{(d) Variance}
\[
\sigma_X^2 = E[X^2]-\mu_X^2 = \frac{2}{\alpha^2}-\frac{1}{\alpha^2} = \frac{1}{\alpha^2}.
\]
So $\sigma_X = 1/\alpha$.

\textbf{(e) Probability within one standard deviation of the mean}
\[
P(\mu_X-\sigma_X \le X \le \mu_X+\sigma_X)
= P\left(0 \le X \le \frac{2}{\alpha}\right)
= F_X\left(\frac{2}{\alpha}\right)-F_X(0)
= 1-e^{-2}.
\]
\[
1-e^{-2} \approx 0.8646647.
\]
\end{stepbox}

\begin{resultsbox}
\[
F_X(x)=\begin{cases}0,&x\le0\\1-e^{-\alpha x},&x>0\end{cases},\quad
E[X]=\frac{1}{\alpha},\quad E[X^2]=\frac{2}{\alpha^2},\quad
\mathrm{Var}(X)=\frac{1}{\alpha^2},
\]
\[
P\left(|X-\mu_X|\le\sigma_X\right)=1-e^{-2}\approx0.8647.
\]
\end{resultsbox}

\question{Exercise 2.8 (Page 76): Sum of Two Zero-Mean Uncorrelated RVs}\label{ex:2.8}
Let $W$ and $V$ be zero-mean, uncorrelated RVs with standard deviations $\sigma_W$ and $\sigma_V$. Define
\[
X = W + V.
\]

\begin{stepbox}
Since $E[W]=E[V]=0$, then $E[X]=0$.

Using uncorrelatedness, $\mathrm{Cov}(W,V)=0$. Therefore,
\[
\mathrm{Var}(X)=\mathrm{Var}(W+V)=\mathrm{Var}(W)+\mathrm{Var}(V)+2\mathrm{Cov}(W,V)
=\sigma_W^2+\sigma_V^2.
\]
Hence,
\[
\sigma_X = \sqrt{\sigma_W^2+\sigma_V^2}.
\]
\end{stepbox}

\begin{resultsbox}
\[
\boxed{\sigma_X = \sqrt{\sigma_W^2 + \sigma_V^2}}
\]
\end{resultsbox}

\question{Exercise 2.12 (Page 76): $Y(t)=X\cos t$}\label{ex:2.12}
Suppose $X$ is a random variable and
\[
Y(t)=X\cos t.
\]

\begin{stepbox}
\textbf{(a) Expected value}
\[
E[Y(t)] = E[X\cos t] = (E[X])\cos t = \bar x\cos t.
\]

\textbf{(b) Time average}
\[
A[Y(t)] = \lim_{T\to\infty}\frac{1}{T}\int_0^T X\cos t\,dt
= X\lim_{T\to\infty}\frac{\sin T}{T}=0.
\]

\textbf{(c) Condition for equality}
For the ensemble mean to match the time average for all $t$,
\[
\bar x\cos t = 0 \quad \forall t \implies \bar x=0.
\]
Also, for an individual realization $y(t)=x\cos t$ to equal its time average for all $t$, one needs $x=0$.
\end{stepbox}

\begin{resultsbox}
\[
E[Y(t)] = \bar x\cos t,\qquad A[Y(t)] = 0.
\]
Mean/time-average equality for all $t$ requires zero mean ($\bar x=0$). For a specific realization to be identically equal to its time average, $x=0$.
\end{resultsbox}
