% --- S2: what the paper is ---
\begin{frame}{What this paper is}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{0.5em}
    \item Each candidate is an executable Python \texttt{compute\_loss} function
    \item An LLM proposes mutations and crossovers over that code
    \item Each variant is scored by actually training agents in Gymnasium
    \item Extends \textbf{REvolve} --- aimed at the \hilite{loss function},
          not the reward
    \item Prompt bans actor--critic, value functions, and Bellman updates
    \item Discovers two new algorithms --- \textbf{CG-FPD} and \textbf{DF-CWP-CP}
    \item Competitive with PPO and SAC across 10 Gymnasium benchmarks
  \end{itemize}
\end{frame}

% --- S3: why automate RL algorithm design ---
\begin{frame}{The canonical RL algorithms}
  \vspace{0.6em}
  \begin{itemize}\setlength\itemsep{0.4em}
    \item \textbf{DQN} (2015) --- value-based, discrete action spaces
    \item \textbf{TRPO} (2015) --- trust-region policy gradient
    \item \textbf{DDPG} (2015) --- off-policy continuous control
    \item \textbf{PPO} (2017) --- clipped policy gradient, the workhorse
    \item \textbf{SAC} (2018) --- entropy-regularized off-policy
    \item \textbf{TD3} (2018) --- twin-critic DDPG
  \end{itemize}
\end{frame}

% --- S4: what has been automated — the ladder ---
\begin{frame}{What AutoML has automated}
  \vspace{0.2em}
  \begin{itemize}\setlength\itemsep{0.25em}
    \item \textbf{Architecture} --- automatically picking network shape, depth, width
      \begin{itemize}\setlength\itemsep{0pt}\item[] \small\color{UofSCGray70}NAS, AmoebaNet, AutoML-Zero\end{itemize}
    \item \textbf{Hyperparameters} --- learning rate, batch size, clipping, $\gamma$
      \begin{itemize}\setlength\itemsep{0pt}\item[] \small\color{UofSCGray70}PB2, OptFormer, PBT\end{itemize}
    \item \textbf{Reward function} --- writing the scalar reward the agent optimizes
      \begin{itemize}\setlength\itemsep{0pt}\item[] \small\color{UofSCGray70}Eureka, REvolve\end{itemize}
    \item \textbf{Preference loss} --- the loss used to learn from pairwise preference data
      \begin{itemize}\setlength\itemsep{0pt}\item[] \small\color{UofSCGray70}Christiano RLHF, DPO, DiscoPOP, PEBBLE\end{itemize}
    \item \hilite{The update rule itself} --- the full RL training loop as code
      \begin{itemize}\setlength\itemsep{0pt}\item[] \small\color{UofSCGray70}\textit{open until this paper}\end{itemize}
  \end{itemize}
\end{frame}

\section*{Results}

% --- S14: two algorithms ---
\begin{frame}{The two algorithms it invented}
  \centering
  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{tikzpicture}
        \node[rectangle, draw=UofSCAtlantic, line width=1.4pt, fill=UofSCGray10,
              inner sep=0.9em, text width=6.3cm, font=\small] {%
          \textbf{\color{UofSCAtlantic}CG-FPD}\\[0.3em]
          \textbf{Confidence-Guided Forward Policy Distillation}\\[0.6em]
          Distill short-horizon plans from a learned latent dynamics model.
          Planning acts as the supervised teacher signal.\\[0.4em]
          \textit{\color{UofSCGray70}No value function. No policy gradient. No Bellman.}};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{tikzpicture}
        \node[rectangle, draw=UofSCCongaree, line width=1.4pt, fill=UofSCGray10,
              inner sep=0.9em, text width=6.3cm, font=\small] {%
          \textbf{\color{UofSCCongaree}DF-CWP-CP}\\[0.3em]
          \textbf{Differentiable Forward Confidence-Weighted Planning with Controllability Prior}\\[0.6em]
          Differentiable world-model rollouts + confidence-weighted desirability + controllability prior.\\[0.4em]
          \textit{\color{UofSCGray70}Also avoids every canonical RL mechanism.}};
      \end{tikzpicture}
    \end{column}
  \end{columns}

  \vspace{1.2em}
  {\large Evolution rediscovered \hilite{model-based planning} --- from scratch.}
\end{frame}

\section*{Discussion}

% --- S18: strengths / weaknesses ---
\begin{frame}{Strengths vs. weaknesses}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\color{UofSCGarnet}Strengths}
      \vspace{0.3em}
      \begin{itemize}
        \item First \hilite{full update-rule} discovery via LLM
        \item Genuinely novel mechanisms (no value, no PG)
        \item Clean extension of REvolve's machinery
        \item LLM-HPO handles fitness fragility cleanly
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\color{UofSCRose}Weaknesses}
      \vspace{0.3em}
      \begin{itemize}
        \item 30\,h per candidate on 4$\times$A100
        \item Narrow benchmarks (CartPole / MuJoCo only)
        \item Is GPT-5.2 \textit{discovering} or \textit{remembering}?
        \item Only 2 evolutionary seeds per LLM
        \item No head-to-head vs.\ DiscoRL
        \item Constitutional constraints never ablated
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
