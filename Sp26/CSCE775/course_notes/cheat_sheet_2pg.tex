\section{Notation and Symbols}

\subsection{Variables and Sets \catnotation}
{\small
\begin{tabular}{@{}>{\color{Atlantic}}l l@{}}
$t$ & Time step index ($t = 0, 1, 2, \ldots$) \\
$S_t, s, s'$ & State (RV), particular values; $s'$ = next state \\
$A_t, a, a'$ & Action (RV), particular values \\
$R_t, r$ & Reward (RV), particular value \\
$G_t$ & Return (cumulative discounted reward) from $t$ \\
$T$ & Terminal time step (episodic tasks) \\
$\gamma$ & Discount factor, $0 \leq \gamma \leq 1$ \\
$\alpha$ & Step-size / learning rate \\
$\theta$ & Convergence threshold \\
$\Delta$ & Max value change across a sweep \\
$k$ & Iteration index \\
$\mathcal{S}, \mathcal{S}^+$ & State set; with terminal state(s) \\
$\mathcal{A}, \mathcal{A}(s)$ & Action set; actions available in $s$ \\
$\mathcal{R}$ & Reward set, $\mathcal{R} \subset \mathbb{R}$ \\
\end{tabular}
}

\subsection{Policies \catnotation}
{\small
\begin{tabular}{@{}>{\color{Atlantic}}l l@{}}
$\pi$ & Policy (states $\to$ action probabilities) \\
$\pi(a \mid s)$ & Prob.\ of action $a$ in state $s$ under $\pi$ \\
$\pi(s)$ & Deterministic policy action in $s$ \\
$\pi_*$ & Optimal policy \\
$\pi'$ & Improved policy \\
\end{tabular}
}

\subsection{Value Functions \catnotation}
{\small
\begin{tabular}{@{}>{\color{Atlantic}}l l@{}}
$v_\pi(s)$ & State-value: expected return from $s$ under $\pi$ \\
$q_\pi(s, a)$ & Action-value: expected return from $s$, taking $a$, under $\pi$ \\
$v_*(s)$ & Optimal state-value: $\max_\pi v_\pi(s)$ \\
$q_*(s, a)$ & Optimal action-value: $\max_\pi q_\pi(s, a)$ \\
$V(s), V_k(s)$ & Estimate of value function; at iteration $k$ \\
\end{tabular}
}

\subsection{Probability and Dynamics \catnotation}
{\small
\begin{tabular}{@{}>{\color{Atlantic}}l l@{}}
$\mathbb{E}_\pi[\cdot]$ & Expected value under policy $\pi$ \\
$p(s', r \mid s, a)$ & Dynamics: $\Pr\{S_t\!=\!s', R_t\!=\!r \mid S_{t-1}\!=\!s, A_{t-1}\!=\!a\}$ \\
$p(s' \mid s, a)$ & State-transition prob (marginalized over $r$) \\
$r(s, a)$ & Expected reward for state-action pair \\
$r(s, a, s')$ & Expected reward for state-action-next-state \\
\end{tabular}
}

\subsection{Operators \catshort}
{\small
\begin{tabular}{@{}>{\color{Horseshoe}}l l@{}}
$\sum_a$ & Shorthand for $\sum_{a \in \mathcal{A}(s)}$ \\
$\sum_{s', r}$ & Shorthand for $\sum_{s' \in \mathcal{S}} \sum_{r \in \mathcal{R}}$ \\
$\max_a$ & Maximize over $a \in \mathcal{A}(s)$ \\
$\arg\max_a$ & Action achieving the maximum \\
$\leftarrow$ & Assignment (update stored value) \\
$\geq$ (policies) & $\pi \geq \pi'$: $v_\pi(s) \geq v_{\pi'}(s)$ for all $s$ \\
\end{tabular}
}

\section{Temporal-Difference Learning \catalgo}
{\small
\colorbox{LightCongaree}{$V(S_t) \leftarrow V(S_t) + \alpha\left[V(S_{t+1}) - V(S_t)\right]$}

Combines Monte Carlo (learning from experience) and DP (bootstrapping from estimates).
}

\section{Finite MDPs}

\subsection{Agent-Environment Interface \catdef}
{\small
Trajectory: $S_0, A_0, R_1, S_1, A_1, R_2, S_2, A_2, R_3, \ldots$

At each $t$: observe $S_t \in \mathcal{S}$, select $A_t \in \mathcal{A}(s)$, receive $R_{t+1} \in \mathcal{R}$, transition to $S_{t+1}$.
}

\subsection{Dynamics Function \catdef}
{\small
\vspace{-4pt}
\colorbox{LightGarnet}{$\displaystyle p(s',r|s,a) = \Pr\{S_t\!=\!s', R_t\!=\!r \mid S_{t-1}\!=\!s, A_{t-1}\!=\!a\}$}
\vspace{2pt}

Must satisfy: $\sum_{s'} \sum_{r} p(s',r|s,a) = 1$ for all $s, a$.
}

\subsection{Derived Quantities \catshort}
{\small
\vspace{-2pt}
\begin{align*}
{\color{Horseshoe}p(s'|s,a)} &= \sum_{r} p(s',r|s,a) \\
{\color{Horseshoe}r(s,a)} &= \sum_{r} r \sum_{s'} p(s',r|s,a) \\
{\color{Horseshoe}r(s,a,s')} &= \sum_{r} r \frac{p(s',r|s,a)}{p(s'|s,a)}
\end{align*}
}

\subsection{Markov Property \catthm}
{\small
Future depends only on current state:
\vspace{-4pt}
\colorbox{LightHoneycomb}{$\displaystyle\Pr\{S_t\!=\!s', R_t\!=\!r \mid S_0, A_0, \ldots, S_{t-1}, A_{t-1}\} = p(s',r|S_{t-1}, A_{t-1})$}
}

\subsection{Returns \catdef}
{\small
\textbf{Episodic}: $G_t = R_{t+1} + R_{t+2} + \cdots + R_T$

\textbf{Continuing (discounted)}: $G_t = \sum_{k=0}^{\infty} \gamma^k R_{t+k+1}$

\textbf{Recursive}: $G_t = R_{t+1} + \gamma G_{t+1}$

\textbf{Constant reward}: $G_t = \sum_{k=0}^{\infty} \gamma^k = \frac{1}{1-\gamma}$
}

\subsection{Policies \catdef}
{\small
$\pi(a|s) = \Pr\{A_t\!=\!a \mid S_t\!=\!s\}$

Deterministic: $\pi(s) = a$.
}

\subsection{Value Functions \catdef}
{\small
\vspace{-2pt}
\begin{align*}
{\color{Garnet}v_\pi(s)} &= \mathbb{E}_\pi[G_t \mid S_t\!=\!s] \\
{\color{Garnet}q_\pi(s,a)} &= \mathbb{E}_\pi[G_t \mid S_t\!=\!s, A_t\!=\!a] \\
v_\pi(s) &= \textstyle\sum_{a} \pi(a|s) \, q_\pi(s,a)
\end{align*}
}

\subsection{Bellman Equations \catkey}
{\small
\vspace{-2pt}
\colorbox{LightRose}{%
\parbox{0.93\columnwidth}{%
\vspace{2pt}
\begin{align*}
v_\pi(s) &= \sum_{a} \pi(a|s) \sum_{s',r} p(s',r|s,a)\big[r + \gamma v_\pi(s')\big] \\
q_\pi(s,a) &= \sum_{s',r} p(s',r|s,a)\big[r + \gamma \sum_{a'} \pi(a'|s') q_\pi(s',a')\big]
\end{align*}
\vspace{-8pt}
}}
}

\subsection{Optimal Value Functions \catdef}
{\small
\vspace{-2pt}
\begin{align*}
{\color{Garnet}v_*(s)} &= \max_\pi v_\pi(s) & {\color{Garnet}q_*(s,a)} &= \max_\pi q_\pi(s,a) \\
v_*(s) &= \max_{a} q_*(s,a) & q_*(s,a) &= \sum_{s',r} p(s',r|s,a)\big[r + \gamma v_*(s')\big]
\end{align*}
}

\subsection{Bellman Optimality Equations \catkey}
{\small
\vspace{-2pt}
\colorbox{LightRose}{%
\parbox{0.93\columnwidth}{%
\vspace{2pt}
\begin{align*}
v_*(s) &= \max_{a} \sum_{s',r} p(s',r|s,a)\big[r + \gamma v_*(s')\big] \\
q_*(s,a) &= \sum_{s',r} p(s',r|s,a)\big[r + \gamma \max_{a'} q_*(s',a')\big]
\end{align*}
\vspace{-8pt}
}}
}

\subsection{Optimal Policy \catdef}
{\small
\vspace{-2pt}
\begin{align*}
\pi_*(s) &= \arg\max_{a} q_*(s,a) \\
         &= \arg\max_{a} \sum_{s',r} p(s',r|s,a)\big[r + \gamma v_*(s')\big]
\end{align*}
\vspace{-6pt}

Any policy greedy w.r.t.\ $v_*$ or $q_*$ is optimal.
}

\section{Dynamic Programming}

{\small DP requires perfect model: $p(s',r|s,a)$.}

\subsection{Policy Evaluation (Prediction) \catalgo}
{\small
Goal: compute $v_\pi$ for given $\pi$. Iterative update:
\vspace{-4pt}
\colorbox{LightCongaree}{$\displaystyle V_{k+1}(s) = \sum_{a} \pi(a|s) \sum_{s',r} p(s',r|s,a)\big[r + \gamma V_k(s')\big]$}
\vspace{2pt}

Stop when $\max_{s} |V_{k+1}(s) - V_k(s)| < \theta$. As $k \to \infty$, $V_k \to v_\pi$.
}

\subsection{Policy Improvement Theorem \catthm}
{\small
\colorbox{LightHoneycomb}{If $q_\pi(s, \pi'(s)) \geq v_\pi(s)$ for all $s$, then $v_{\pi'}(s) \geq v_\pi(s)$ for all $s$.}
}
