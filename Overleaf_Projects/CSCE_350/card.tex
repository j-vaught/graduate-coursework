\section*{Definitions}

\subsubsection*{Big O}
For functions \( f(n) \) and \( g(n) \), \( f(n) = O(g(n)) \) if there exist constants \( c > 0 \) and \( n_0 \) such that \( 0 \leq f(n) \leq c \cdot g(n) \) for all \( n \geq n_0 \).

\subsubsection*{Theta}
\( f(n) = \Theta(g(n)) \) if and only if \( f(n) = O(g(n)) \) and \( f(n) = \Omega(g(n)) \).

\subsubsection*{Omega}
For functions \( f(n) \) and \( g(n) \), \( f(n) = \Omega(g(n)) \) if there exist constants \( c > 0 \) and \( n_0 \) such that \( 0 \leq c \cdot g(n) \leq f(n) \) for all \( n \geq n_0 \).
