\documentclass[border=10pt]{standalone}
\usepackage[american]{circuitikz}

\begin{document}
\begin{circuitikz}[american, scale=1.0, every node/.style={font=\small}]

    % Draw the AC voltage source
    \draw (-4, 1) to[sV=$V$] ++(0, -2); % Voltage source symbol
    \draw (-4,1) -- (-1.7,1);          % Wire from voltage source positive terminal
    \draw (-4,-1) -- (-1.7,-1);        % Wire from voltage source negative terminal
    \node at (-5.2, 0) {120 VAC};      % Label for AC voltage source

    % Draw the transformer
    \draw (-1.7,1) to[L] (-1.7,-1);    % Primary coil (high voltage side)
    \draw (-1.3,1) -- (-1.3,-1);       % Transformer core (right)
    \draw (-1.4,1) -- (-1.4,-1);       % Transformer core (left)
    \draw (-1,-1) to[L] (-1,1);        % Secondary coil (low voltage side)

    % Connect transformer secondary to bridge rectifier inputs
    \draw (-1,1) -- (1,1);             % Wire from transformer secondary (positive side)
    \draw (-1,-1) -- (1,-1);           % Wire from transformer secondary (negative side)

    % Draw the full-wave bridge rectifier
    \ctikzset{diodes/scale=0.5}        % Adjust diode size for clarity
    \draw (1,-1) to[D] ++(1,1);        % Diode 1 (negative to positive path)
    \draw (1,1) to[D] ++(1,-1);        % Diode 2 (positive to negative path)
    \draw (0,0) to[D] ++(1,1);         % Diode 3 (positive output)
    \draw (0,0) to[D] ++(1,-1);        % Diode 4 (negative output)

    % Draw wires from rectifier to the capacitor and load
    % Negative rail
    \draw (0,0) -- (-0.2,0); 
    \draw (-0.2,0) -- (-0.2,-1.4); 
    \draw (-0.2,-1.4) -- (2.2,-1.4);
    \draw (2.2,-1.4) -- (2.2,-1); 
    \draw (2.2,-1) -- (4,-1);

    % Positive rail
    \draw (2,0) -- (2.2, 0); 
    \draw (2.2,0) -- (2.2, 1);
    \draw (2.2,1) -- (4, 1);

    % Draw the capacitor
    \draw (4,1) to[C, l=$C$] (4,-1); % Capacitor between positive and negative rails

    % Connect the load resistor to the capacitor
    \draw (4,1) -- (6,1);             % Wire from capacitor positive to resistor
    \draw (6,1) to[R, l=$R$] (6,-1);  % Load resistor
    \draw (6,-1) -- (4,-1);           % Wire back to capacitor negative

\end{circuitikz}
\end{document}
