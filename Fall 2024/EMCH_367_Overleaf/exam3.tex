\section*{Given Problem}
Consider the transfer function:
\begin{equation}
G(s) = \frac{(s+1)(s+5)}{s(s^2 +4s +10)(s^2 +8s +25)},
\end{equation}
and the input:
\begin{equation}
u(t) = \sin(2t -1).
\end{equation}
We wish to find the steady-state output:
\begin{equation}
y_{ss}(t).
\end{equation}

\section*{Solution}

First, expanding the numerator:
\begin{equation}
(s+1)(s+5) = s^2 + 6s +5.
\end{equation}
Thus,
\begin{equation}
G(s) = \frac{s^2 +6s +5}{s(s^2 +4s +10)(s^2 +8s +25)}.
\end{equation}
Expanding the denominator polynomials:
\begin{equation}
(s^2 +4s +10)(s^2 +8s +25) = s^4 +12s^3 +67s^2 +180s +250.
\end{equation}
Multiplying by $s$:
\begin{equation}
s(s^4 +12s^3 +67s^2 +180s +250) = s^5 +12s^4 +67s^3 +180s^2 +250s.
\end{equation}
Therefore, the full denominator is:
\begin{equation}
s^5 +12s^4 +67s^3 +180s^2 +250s.
\end{equation}
So:
\begin{equation}
G(s) = \frac{s^2 +6s +5}{s^5 +12s^4 +67s^3 +180s^2 +250s}.
\end{equation}
Substituting $s = j\omega$

\begin{equation}
G(s) = \frac{(j\omega)^2 +6(j\omega) +5}{(j\omega)^5 +12(j\omega)^4 +67(j\omega)^3 +180(j\omega)^2 +250(j\omega)}.
\end{equation}
Simplifying all imaginary numbers,
\begin{equation}
G(s) = \frac{-\omega^2 +6j\omega +5}{j\omega^5 +12\omega^4 -67j\omega^3 -180\omega^2 +250j\omega}.
\end{equation}
Then, computing magnitude via
\begin{equation}
|H(j)| = \sqrt{(Im)^2+(Re)^2},
\end{equation}
and particularly
\begin{equation}
|H(j)| = \frac{\sqrt{(Im)_{num}^2+(Re)_{num}^2}}{\sqrt{(Im)_{den}^2+(Re)_{den}^2}}.
\end{equation}
we get
\begin{equation}
H(j\omega)_{num} = (-2)^2 +6j*2 +5.
\end{equation}
\begin{flushright}\color{red}\scriptsize \textbf{Real numbers should be combined here to 1, not 4 and 5.}\end{flushright}

\begin{equation}
H(j\omega)_{den} = -j*2^5 +12*2^4 + 67j*2^3 +180*2^2 -250j*2.
\end{equation}
\begin{flushright}\color{red}\scriptsize \textbf{Denominator terms are also combined incorrectly.}\end{flushright}
Finding magnitude, we get:
\begin{equation}
|H(j\omega)_{num}| = \sqrt{4^2+12^2+5^2}=\sqrt{185}.
\end{equation}
\begin{flushright}\color{red}\scriptsize \textbf{This would be correct except that I squared all terms instead of real/imag. [Mistake carried from Eq. 14]}\end{flushright}

\begin{equation}
|H(j\omega)_{den}| = \sqrt{-1024 + 36,864 + 287,296 + 518,400 - 250,000}=\sqrt{591,536}.
\end{equation}
\begin{flushright}\color{red}\scriptsize \textbf{The denominator arithmetic is incorrect [Mistake carried from Eq. 15]}\end{flushright}
Thus, 
\begin{equation}
|H(j\omega)| = \frac{\sqrt{185}}{\sqrt{591,536}} \approx 0.017685.
\end{equation}
Then, for angle using
\begin{equation}
\angle H(j\omega) = \tan^{-1}\left(\frac{Im}{Re}\right),
\end{equation}
and
\begin{equation}
\angle H(j\omega) = \angle H(j\omega)_{num} - \angle H(j\omega)_{den},
\end{equation}
we get:
\begin{equation}
\angle H(j\omega)_{num} = \tan^{-1}\left(\frac{12}{1}\right) \approx 1.48766 \text{ rad}.
\end{equation}


\begin{equation}
\angle H(j\omega)_{den} = \tan^{-1}\left(\frac{-4}{-528}\right) \approx 0.007576 \text{ rad}.
\end{equation}
\begin{flushright}\color{red}\scriptsize \textbf{The denominator's real+imag parts are negative, needing to add a $\pi$ to the angle.}\end{flushright}
Finally, 
\begin{equation}
\angle H(j\omega) = 1.48766 - 0.007576 \approx 1.48 \text{ rad}.
\end{equation}
Putting into FINAL form, 
\begin{equation}
y_{ss}(t) = 0.017685*\sin(2t-1+1.48 \text{ rad}).
\end{equation}
