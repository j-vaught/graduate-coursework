% ----- URL display helper
\usepackage{xstring}
\newcommand{\DisplayURL}[1]{%
  \begingroup
    \def\base{#1}%
    \StrSubstitute{\base}{https://}{}[\base]%
    \StrSubstitute{\base}{http://}{}[\base]%
    \StrSubstitute{\base}{www.}{}[\base]%
    \IfEndWith{\base}{/}{\StrGobbleRight{\base}{1}[\base]}{}%
    \texttt{\base}%
  \endgroup
}
