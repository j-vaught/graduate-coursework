% ----- URL display helper (strips https://www.)
\newcommand{\DisplayURL}[1]{%
  \begingroup
    \def\base{#1}%
    \StrSubstitute{\base}{https://}{}[\base]%
    \StrSubstitute{\base}{http://}{}[\base]%
    \StrSubstitute{\base}{www.}{}[\base]%
    \IfEndWith{\base}{/}{\StrGobbleRight{\base}{1}[\base]}{}%
    \texttt{\base}%
  \endgroup
}

% ----- Reference ID overlay (always bottom-right)
\newcommand{\PlaceRefID}{%
  \stepcounter{flyercount}%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=south east, text=dgLightGray] at ($(current page.south east)+(-0.4in,0.15in)$) {%
      \fontsize{\RefIDFontSize}{12}\selectfont

%========================================================================
% CENTERED LAYOUT - All elements at fixed positions
%========================================================================
\newcommand{\MakeTVCentered}[2]{%
  \thispagestyle{empty}%
  \begin{tikzpicture}[remember picture,overlay]
    %
    % --- Headline (top, centered, fixed position) ---
    \node[anchor=south] at ($(current page.south)+(0,\HeadlineY)$) {%
      \begin{minipage}{15in}%
        \centering
        \fontsize{\HeadlineSize}{\HeadlineLeading}\selectfont\bfseries
        #1\par
      \end{minipage}%
    };
    %
    % --- Line under headline (fixed position) ---
    \draw[line width=\RuleThickness]
      ($(current page.south west)+(0.3in,\TopLineY)$) -- ($(current page.south east)+(-0.3in,\TopLineY)$);
    %
    % --- Middle content: QR + Description (fixed center position) ---
    \node[anchor=center] at ($(current page.south)+(0,\MiddleY)$) {%
      \begin{minipage}{15in}%
        \centering
        % QR Code on left
        \begin{minipage}[c]{0.38\linewidth}%
          \centering
          \QRCodeOnly\\[\QRGap]
          {\fontsize{\QRURLSize}{22}\selectfont \href{\WriteURL}{\DisplayURL{\WriteURL}}}
        \end{minipage}%
        \hspace{0.06\linewidth}%
        % Description text on right
        \begin{minipage}[c]{0.5\linewidth}%
          \centering
          \fontsize{\DescSize}{\DescLeading}\selectfont
          #2\\[0.5em]
          \PublishPunch\par
        \end{minipage}%
      \end{minipage}%
    };
    %
    % --- Bottom line (fixed position) ---
    \draw[line width=\RuleThickness]
      ($(current page.south west)+(0.3in,\BottomLineY)$) -- ($(current page.south east)+(-0.3in,\BottomLineY)$);
    %
    % --- Logo (fixed position) ---
    \node[anchor=south] at ($(current page.south)+(0,\LogoY)$) {%
      \includegraphics[width=3in]{DG_logo.pdf}%
    };
  \end{tikzpicture}%
  \PlaceRefID
  \newpage
}
