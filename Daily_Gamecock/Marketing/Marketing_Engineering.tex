% ----- Headline styles
\newcommand{\FlyerHeadline}[1]{%
  \begin{center}%
    \fontsize{\HeadlineSize}{\HeadlineLeading}\selectfont\bfseries
    #1\unskip\par % \unskip eats accidental spaces at end of input
  \end{center}%
}

\newcommand{\FlyerSubhead}[1]{%
  \begin{center}%
    \vspace{0em}%
    \fontsize{\SubheadSize}{\SubheadLeading}\selectfont
    #1\unskip\SubheadSuffix\par % \unskip ensures no gap before suffix
  \end{center}%
}

  \vspace{\RuleVSpace}%
}

% ----- URL display helper
\usepackage{xstring}
\newcommand{\DisplayURL}[1]{%
  \begingroup
    \def\base{#1}%
    \StrSubstitute{\base}{https://}{}[\base]%
    \StrSubstitute{\base}{http://}{}[\base]%
    \StrSubstitute{\base}{www.}{}[\base]%
    \IfEndWith{\base}{/}{\StrGobbleRight{\base}{1}[\base]}{}%
    \texttt{\base}%
  \endgroup
}

% ----- QR block
\newcommand{\QRBlock}{%
  \begin{center}%
    {\fontsize{\QRLabelSize}{\QRLabelLeading}\selectfont\bfseries \QRLabelText}\par
    \vspace{\QRGap}%
    \qrcode[height=\QRSize]{\WriteURL}\par
    \vspace{\QRGap}%
    {\fontsize{\QRURLSize}{\QRURLLeading}\selectfont \href{\WriteURL}{\DisplayURL{\WriteURL}}}%
  \end{center}%
}

% ----- Overlay QR placement
\newcommand{\PlaceQR}{%
  \begin{tikzpicture}[remember picture,overlay]
    \node at ($(current page.center)+(\QRXOffset,\QRYOffset)$) {%
      \begin{minipage}{0.6\linewidth}
        \centering
        \QRBlock
      \end{minipage}
    };
  \end{tikzpicture}%
}

% ----- Reference ID overlay (takes reference code as parameter)
\newcommand{\PlaceRefID}[1]{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=south east, text=dgLightGray] at ($(\RefCorner)+(\RefXOffset,\RefYOffset)$) {%
      \fontsize{\RefIDFontSize}{7}\selectfont
      #1 ~|~ Remove after: \RemoveDateText
    };
  \end{tikzpicture}%
}
