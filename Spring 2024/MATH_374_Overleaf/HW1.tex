\section*{Problem 2}
Given the truth values \( A = \text{true} \), \( B = \text{false} \), and \( C = \text{true} \), what is the truth value of each of the following wffs?
\begin{enumerate}
    \item \( A \land (B \lor C) \)
    \item \( (A \land B) \lor C \)
    \item \( (A \land B)' \lor C \)
    \item \( A' \lor (B' \land C') \)
\end{enumerate}
\newpage
\subsection*{Solution}
Given the truth values \( A = \text{true} \), \( B = \text{false} \), and \( C = \text{true} \), we are tasked with finding the truth value of each of the following well-formed formulas (wffs):

\begin{enumerate}
    \item \( A \land (B \lor C) \)
    \begin{itemize}
        \item First, evaluate \( B \lor C \) which is \( \text{false} \lor \text{true} = \text{true} \).
        \item Next, evaluate \( A \land (B \lor C) \), substituting \( \text{true} \land \text{true} \) which results in \( \text{true} \).
    \end{itemize}

    \item \( (A \land B) \lor C \)
    \begin{itemize}
        \item First, evaluate \( A \land B \) which is \( \text{true} \land \text{false} = \text{false} \).
        \item Next, evaluate \( (A \land B) \lor C \), substituting \( \text{false} \lor \text{true} \) which results in \( \text{true} \).
    \end{itemize}

    \item \( (A \land B)' \lor C \)
    \begin{itemize}
        \item First, evaluate \( A \land B \) which is \( \text{true} \land \text{false} = \text{false} \).
        \item Then, negate \( (A \land B) \) to get \( (A \land B)' = \text{true} \).
        \item Finally, evaluate \( (A \land B)' \lor C \) which is \( \text{true} \lor \text{true} = \text{true} \).
    \end{itemize}

    \item \( A' \lor (B' \land C') \)
    \begin{itemize}
        \item First, negate \( A \) to get \( A' = \text{false} \).
        \item Then, negate \( B \) and \( C \) to get \( B' = \text{true} \) and \( C' = \text{false} \).
        \item Next, evaluate \( B' \land C' \) which is \( \text{true} \land \text{false} = \text{false} \).
        \item Finally, evaluate \( A' \lor (B' \land C') \) which is \( \text{false} \lor \text{false} = \text{false} \).
    \end{itemize}
\end{enumerate}
By following the above steps, the truth values for the wffs are as follows:
\begin{itemize}
    \item \( A \land (B \lor C) \) is \text{true}.
    \item \( (A \land B) \lor C \) is \text{true}.
    \item \( (A \land B)' \lor C \) is \text{true}.
    \item \( A' \lor (B' \land C') \) is \text{false}.
\end{itemize}

\section*{Problem 7}
Write the negation of each statement.
\begin{enumerate}
    \item \( \text{If the food is good (A), then the service is excellent (B)} \)
    \item \( \text{Either the food is good (A) or the service is excellent (B)} \)
\end{enumerate}
\newpage
\subsection*{Solution}
Let \( A \) represent "food is good" and \( B \) represent "service is excellent."
\begin{enumerate}
    \item \( A \Rightarrow B \)
    \begin{itemize}
        \item Negation: \( A \land B' \)
        \item In words: The food is good and the service is not excellent.
    \end{itemize}
    
    \item \( A \lor B \)
    \begin{itemize}
        \item Negation: \( A' \land B' \)
        \item In words: The food is not good and the service is not excellent.
    \end{itemize}
\end{enumerate}

\section*{Problem 8}
Write the negation of each statement.
\begin{enumerate}
    \item \( \text{The processor is fast but the printer is slow} \)
    \item \( \text{The printer is slow only if the file is damaged} \)
\end{enumerate}
\newpage
\subsection*{Solution}
Let \( A \) represent "Processor is fast," \( B \) represent "Printer is slow," and \( C \) represent "File is Damaged."
\begin{enumerate}
    \item \( A \land B \)
    \begin{itemize}
        \item Negation: \( A' \lor B' \)
        \item In words: The processor is not fast or the printer is not slow.
    \end{itemize}
    
    \item \( B \Rightarrow C \)
    \begin{itemize}
        \item Negation: \( B \land C' \)
        \item In words: The printer is slow and the file is not damaged.
    \end{itemize}
\end{enumerate}

\section*{Problem 9}
Let \( A \), \( B \), and \( C \) be the following statements:
\begin{align*}
    A & : \text{Roses are red} \\
    B & : \text{Violets are blue} \\
    C & : \text{Sugar is sweet}
\end{align*}
Translate the following compound statements into symbolic notation.
\begin{enumerate}
    \item \( \text{Roses are red and violets are blue} \)
    \item \( \text{Roses are red, and either violets are blue or sugar is sweet} \)
    \item \( \text{Whenever violets are blue, roses are red and sugar is sweet} \)
    \item \( \text{Roses are red only if violets aren’t blue or sugar is sour} \)
    \item \( \text{Roses are red and, if sugar is sour, then either violets aren’t blue or sugar is sweet} \)
\end{enumerate}
\newpage
\subsection*{Solution}
\begin{enumerate}
    \item \( \text{Roses are red and violets are blue} \)
    \begin{itemize}
        \item Symbolic Notation: \( A \land B \)
    \end{itemize}

    \item \( \text{Roses are red, and either violets are blue or sugar is sweet} \)
    \begin{itemize}
        \item Symbolic Notation: \( A \land (B \lor C) \)
    \end{itemize}

    \item \( \text{Whenever violets are blue, roses are red and sugar is sweet} \)
    \begin{itemize}
        \item Symbolic Notation: \( B \Rightarrow (A \land C) \)
    \end{itemize}

    \item \( \text{Roses are red only if violets aren't blue or sugar is sour} \)
    \begin{itemize}
        \item Symbolic Notation: \( A \Rightarrow (B' \lor C') \)
    \end{itemize}

    \item \( \text{Roses are red and, if sugar is sour, then either violets aren't blue or sugar is sweet} \)
    \begin{itemize}
        \item Symbolic Notation: \( A \land (C' \Rightarrow (B' \lor C)) \)
    \end{itemize}
\end{enumerate}

\section*{Problem 11}
Use \( A \), \( B \), and \( C \) as defined in Exercise 9 to translate the following statements into English.
\begin{enumerate}
    \item \( B \lor C' \)
    \item \( B' \lor (A \rightarrow C) \)
    \item \( (C \land A) \rightarrow B \)
\end{enumerate}
\newpage
\subsection*{Solution}
\begin{enumerate}
    \item \( B \lor C' \)
    \begin{itemize}
        \item English Translation: Either violets are blue, or sugar is not sweet (sour).
    \end{itemize}
    
    \item \( B' \lor (A \rightarrow C) \)
    \begin{itemize}
        \item English Translation: Either violets are not blue, or if roses are red, then sugar is sweet.
    \end{itemize}
    
    \item \( (C \land A) \rightarrow B \)
    \begin{itemize}
        \item English Translation: If sugar is sweet and roses are red, then violets are blue.
    \end{itemize}
\end{enumerate}
